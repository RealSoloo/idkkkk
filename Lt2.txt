local function axeDamage(name, class)
	local a = AXES[name]
	if not a then return 1 end
	local v = class and a[class]
	if v == nil then v = a[1] end
	return v
end
local function axeNameOf(tool)
	local v = tool:FindFirstChild("ToolTip")
	if v and v:IsA("StringValue") then return v.Value end
	local ok, tip = pcall(function() return tool.ToolTip end)
	if ok and type(tip) == "string" then return tip end
	return nil
end
local function bestAxe(class)
	local best, bestName, bestScore
	local function try(tool)
		if not tool:IsA("Tool") or tool.Name == "BlueprintTool" then return end
		local nm = axeNameOf(tool)
		if not nm or nm == "" then return end
		local low = nm:lower()
		if not AXES[nm] and not low:find("axe", 1, true) and not low:find("hatchet", 1, true) then return end
		local score = axeDamage(nm, class)
		if not bestScore or score > bestScore then best, bestName, bestScore = tool, nm, score end
	end
	local ch = getChar()
	if ch then for _, c in ipairs(ch:GetChildren()) do try(c) end end
	local bp = LocalPlayer:FindFirstChildOfClass("Backpack")
	if bp then for _, c in ipairs(bp:GetChildren()) do try(c) end end
	return best, bestName, bestScore
end

-- ---------------------------------------------------- tree / log helpers
local function regions()
	local out = {}
	for _, f in ipairs(Workspace:GetChildren()) do
		if f.Name:lower():find("treeregion", 1, true) then table.insert(out, f) end
	end
	return out
end
local function classOf(tree)
	local tc = tree:FindFirstChild("TreeClass")
	return tc and tc:IsA("StringValue") and tc.Value or nil
end
local function sectionsOf(tree)
	local s = {}
	for _, p in ipairs(tree:GetChildren()) do
		if p.Name == "WoodSection" and p:IsA("BasePart") then table.insert(s, p) end
	end
	table.sort(s, function(a, b) return a.Position.Y < b.Position.Y end)
	return s
end
local function baseSection(tree)
	local list = sectionsOf(tree)
	local base = list[1]
	for _, s in ipairs(list) do
		local id = s:FindFirstChild("ID")
		if id and id.Value == 1 then base = s break end
	end
	return base
end
local function scanClasses()
	local seen, list = {}, {}
	for _, f in ipairs(regions()) do
		for _, m in ipairs(f:GetChildren()) do
			if m:IsA("Model") then
				local c = classOf(m)
				if c and not seen[c] and #sectionsOf(m) > 1 then seen[c] = true table.insert(list, c) end
			end
		end
	end
	table.sort(list)
	LT.classes = list
	return list
end
-- class == nil means any class. mode 1 = most sections, 2 = closest to you.
local function pickTree(class, mode)
	local r = getRoot()
	local best, bestV
	for _, f in ipairs(regions()) do
		for _, m in ipairs(f:GetChildren()) do
			if m:IsA("Model") and (class == nil or classOf(m) == class) and classOf(m) then
				local n = #sectionsOf(m)
				local skip = n < 2 or (mode == 1 and classOf(m) == "Generic" and n < 12)
				if not skip then
					local v
					if mode == 1 and class ~= nil then
						v = -n
					else
						local ok, pv = pcall(function() return m:GetPivot().Position end)
						v = (ok and r) and (pv - r.Position).Magnitude or math.huge
					end
					if not bestV or v < bestV then best, bestV = m, v end
				end
			end
		end
	end
	return best
end
local function ownedByMe(model)
	local o = model:FindFirstChild("Owner")
	if o then
		if o:IsA("ObjectValue") and o.Value == LocalPlayer then return true end
		if o:IsA("StringValue") and o.Value == LocalPlayer.Name then return true end
		local os2 = o:FindFirstChild("OwnerString")
		if os2 and os2:IsA("StringValue") and os2.Value == LocalPlayer.Name then return true end
	end
	local s = model:FindFirstChild("OwnerString")
	return s ~= nil and s:IsA("StringValue") and s.Value == LocalPlayer.Name
end
local function myLogs()
	local out = {}
	local lm = Workspace:FindFirstChild("LogModels")
	if lm then for _, m in ipairs(lm:GetChildren()) do if m:IsA("Model") and ownedByMe(m) then table.insert(out, m) end end end
	return out
end
local function snapshotLogs()
	local snap, lm = {}, Workspace:FindFirstChild("LogModels")
	if lm then for _, m in ipairs(lm:GetChildren()) do snap[m] = true end end
	return snap
end
-- new log models of this class that are mine (or whose owner label hasn't loaded yet)
local function newLogs(class, snap)
	local out, lm = {}, Workspace:FindFirstChild("LogModels")
	if not lm then return out end
	for _, m in ipairs(lm:GetChildren()) do
		if not snap[m] and m:IsA("Model") then
			local c = m:FindFirstChild("TreeClass")
			if c and c.Value == class and (m:FindFirstChild("Owner") == nil or ownedByMe(m)) then table.insert(out, m) end
		end
	end
	return out
end

-- ---------------------------------------------------- preview widget (rotating 3D model or an image, with a caption)
local function makePreview(parent, height, caption)
	local f = new("Frame", { Size = UDim2.new(1, 0, 0, height), BackgroundTransparency = 1 }, parent)
	local box = new("Frame", { Size = UDim2.new(1, -28, 1, -26), Position = UDim2.fromOffset(14, 6), BackgroundColor3 = T.Bg, BorderSizePixel = 0, ClipsDescendants = true }, f)
	round(box, 8)
	local vp = new("ViewportFrame", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Ambient = Color3.fromRGB(190, 190, 190), LightColor = Color3.new(1, 1, 1), Visible = false }, box)
	local cam = new("Camera", { FieldOfView = 50 }, vp)
	vp.CurrentCamera = cam
	local img = new("ImageLabel", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, ScaleType = Enum.ScaleType.Fit, Visible = false }, box)
	local cap = new("TextLabel", { Size = UDim2.new(1, -28, 0, 16), Position = UDim2.new(0, 14, 1, -19), BackgroundTransparency = 1, Text = caption, TextColor3 = T.Sub, Font = Enum.Font.Gotham, TextSize = 11, TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd }, f)
	local obj, shown = {}, nil
	local center, dist, rad, ang = Vector3.zero, 10, 5, 0
	local function clear()
		if shown then shown:Destroy() shown = nil end
		vp.Visible = false
		img.Visible = false
	end
	function obj.Clear(text) clear() cap.Text = text or caption end
	function obj.Caption(text) cap.Text = text end
	function obj.Image(asset, text)
		clear()
		img.Image = asset
		img.Visible = true
		cap.Text = text or ""
	end
	function obj.Model(model, text)
		clear()
		cap.Text = text or ""
		if not model then return end
		if LT.previews == false then cap.Text = (text or "") .. "  (preview off)" return end
		local ok, c = pcall(function() return model:Clone() end)
		if not (ok and c) then cap.Text = (text or "") .. "  (no preview)" return end
		for _, d in ipairs(c:GetDescendants()) do
			if d:IsA("BaseScript") or d:IsA("Sound") then d:Destroy() end
		end
		local cf, size
		if c:IsA("BasePart") then
			cf, size = c.CFrame, c.Size
		else
			local okb, a, b = pcall(c.GetBoundingBox, c)
			if okb then cf, size = a, b end
		end
		if not cf then c:Destroy() cap.Text = (text or "") .. "  (no preview)" return end
		c.Parent = vp
		shown = c
		center = cf.Position
		rad = math.max(size.Magnitude / 2, 1)
		dist = rad / math.tan(math.rad(cam.FieldOfView / 2)) * 1.1
		vp.Visible = true
	end
	local conn
	conn = RunService.Heartbeat:Connect(function(dt)
		if not f.Parent then conn:Disconnect() return end
		if not shown or not vp.Visible then return end
		ang = ang + dt * 0.7
		cam.CFrame = CFrame.lookAt(center + Vector3.new(math.cos(ang) * dist, rad * 0.4, math.sin(ang) * dist), center)
	end)
	return obj
end

-- ---------------------------------------------------- status / job runner
local function setStatus(text) if statusRow then statusRow.Text = "Status: " .. text end end
local function dropLock() if LT.lock then LT.lock:Disconnect() LT.lock = nil end end
local function stopAll() LT.token = LT.token + 1 LT.busy = false dropLock() setStatus("Idle") end
local function runJob(label, fn)
	if LT.busy then notify("Already running - press Stop first", T.Warn) return end
	if not RemoteProxy or not DragRemote then
		Interaction = ReplicatedStorage:FindFirstChild("Interaction")
		RemoteProxy = Interaction and Interaction:FindFirstChild("RemoteProxy")
		DragRemote = Interaction and Interaction:FindFirstChild("ClientIsDragging")
	end
	if not RemoteProxy or not DragRemote then notify("LT2 remotes not found - are you in Lumber Tycoon 2?", T.Bad) return end
	LT.token = LT.token + 1
	local my = LT.token
	LT.busy = true
	setStatus(label)
	task.spawn(function()
		local alive = function() return Live.Running and LT.token == my end
		local good, err = pcall(fn, alive)
		if not good then notify("LT2 error: " .. tostring(err):sub(1, 70), T.Bad) end
		if LT.token == my then LT.busy = false dropLock() setStatus("Idle") end
	end)
end
local function waitRespawn(alive)
	local deadline = os.clock() + 30
	repeat task.wait(0.2) until not alive() or os.clock() > deadline or (getHum() and getHum().Health > 0 and getRoot())
	task.wait(1)
end

-- ---------------------------------------------------- cutting
local function cutSection(section, tool, axeName, class, stop)
	if not section or not section.Parent then return end
	local idObj = section:FindFirstChild("ID")
	if not idObj then return end
	local cutEvent = section:FindFirstChild("CutEvent") or section.Parent:FindFirstChild("CutEvent") or (section.Parent.Parent and section.Parent.Parent:FindFirstChild("CutEvent"))
	if not cutEvent then return end
	local sizeY = section.Size.Y
	local args = {
		sectionId = idObj.Value, faceVector = Vector3.new(0, 0, -1),
		height = sizeY * math.clamp(0.1 + (8 - sizeY) / 60, 0.1, 0.2),
		hitPoints = axeDamage(axeName, class), cooldown = 0, cuttingClass = "Axe", tool = tool,
	}
	for _ = 1, LT.fires do
		if not section.Parent or stop() then break end
		RemoteProxy:FireServer(cutEvent, args)
		task.wait(math.max(LT.fireDelay / 100, 0.005))
	end
end

-- ---------------------------------------------------- moving logs (claim ownership, then set CFrame)
local function claimDrag(model, timeout)
	local owner = model:FindFirstChild("Owner")
	local li = (owner and owner:FindFirstChild("LastInteraction")) or model:FindFirstChild("LastInteraction")
	local changed, conn = false, nil
	if li then conn = li:GetPropertyChangedSignal("Value"):Connect(function() changed = true end) end
	local deadline = os.clock() + timeout
	while Live.Running and os.clock() < deadline and not changed do
		pcall(DragRemote.FireServer, DragRemote, model)
		task.wait()
	end
	if conn then conn:Disconnect() end
	return changed
end
local function moveLog(part, goalCF)
	if not part or not part.Parent then return false end
	local model = part:FindFirstAncestorOfClass("Model") or part.Parent
	local r = getRoot()
	if r then
		local flat = (r.Position - part.Position) * Vector3.new(1, 0, 1)
		if flat.Magnitude > 10 then r.CFrame = CFrame.new(part.Position + Vector3.new(0, 3, 0)) task.wait(0.05) end
	end
	claimDrag(model, 1)
	if part.Parent then part.CFrame = goalCF end
	task.wait(0.1)
	return true
end
local function bringParts(parts, alive, center)
	local r = getRoot()
	if not r then return end
	local home = r.CFrame
	center = center or home
	for i, p in ipairs(parts) do
		if not alive() then break end
		local col, row = (i - 1) % 5, math.floor((i - 1) / 5)
		moveLog(p, center * CFrame.new(col * 5, 0, -6 - row * 6))
	end
	local rr = getRoot()
	if rr and LT.back then rr.CFrame = home end
end
local function innerWoods(models)
	local out = {}
	for _, m in ipairs(models) do
		local iw = m:FindFirstChild("InnerWood")
		if iw and iw:IsA("BasePart") then table.insert(out, iw) end
	end
	return out
end

-- ---------------------------------------------------- chop one tree (teleport onto the base, cut until it falls)
local function chopTree(tree, alive)
	local class = classOf(tree)
	if not class then return false end
	local tool, axeName = bestAxe(class)
	if not tool then notify("No axe found in your backpack", T.Warn) return false end
	if showTreePreview then showTreePreview(tree, "Chopping") end
	local r = getRoot()
	if not r or #sectionsOf(tree) == 0 then return false end
	local snap = snapshotLogs()
	local origin = r.CFrame
	local base = baseSection(tree)
	r.CFrame = base.CFrame
	r.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
	dropLock()
	LT.lock = RunService.Heartbeat:Connect(function()
		local rr = getRoot()
		if rr and base and base.Parent then rr.CFrame = base.CFrame rr.AssemblyLinearVelocity = Vector3.new(0, 0, 0) end
	end)
	task.wait(0.1)
	local lastCheck, fellNow = 0, false
	local function fell()
		if fellNow then return true end
		if os.clock() - lastCheck > 0.25 then
			lastCheck = os.clock()
			fellNow = tree.Parent == nil or #newLogs(class, snap) > 0
		end
		return fellNow
	end
	while alive() and not fell() do
		local h = getHum()
		if not h or h.Health <= 0 then
			waitRespawn(alive)
			if not alive() then break end
			tool, axeName = bestAxe(class)
			if not tool then notify("No axe after respawn - stopped", T.Warn) break end
		end
		base = baseSection(tree)
		if not base then break end
		cutSection(base, tool, axeName, class, function() return not alive() or fell() end)
		task.wait(LT.sweep / 10)
	end
	dropLock()
	if not alive() then return false end
	task.wait(0.3)
	local rr = getRoot()
	if rr and LT.back then rr.CFrame = origin end
	if LT.bring then
		local stumps = innerWoods(newLogs(class, snap))
		if #stumps > 0 then setStatus("Bringing logs") bringParts(stumps, alive) end
	end
	return true
end

-- ---------------------------------------------------- selling (sections one by one onto the sell spot)
local function sellLogs(alive)
	local models = myLogs()
	if #models == 0 then notify("You have no logs to sell", T.Warn) return end
	local r = getRoot()
	local home = r and r.CFrame
	local idx = 0
	for _, m in ipairs(models) do
		if not alive() then break end
		local secs = {}
		for _, d in ipairs(m:GetDescendants()) do
			if d:IsA("BasePart") and d.Name == "WoodSection" then table.insert(secs, d) end
		end
		table.sort(secs, function(a, b) return a.Position.Y < b.Position.Y end)
		for _, s in ipairs(secs) do
			if not alive() then break end
			if s.Parent then
				for _, d in ipairs(m:GetDescendants()) do
					if (d:IsA("Weld") or d:IsA("WeldConstraint") or d:IsA("ManualWeld")) and (d.Part0 == s or d.Part1 == s) then pcall(function() d:Destroy() end) end
				end
				task.wait(0.1)
				idx = idx + 1
				local slot = (idx - 1) % LT.sellWrap
				moveLog(s, CFrame.new(LT.sellPos.X + slot * 3, LT.sellPos.Y, LT.sellPos.Z))
			end
		end
	end
	local rr = getRoot()
	if rr and home and LT.back then rr.CFrame = home end
end

local function shownClasses()

showTreePreview = function(tree, prefix)
	if not tree or not tree.Parent then treePrev.Clear("No tree found") LT.prevTree = nil return end
	local class = classOf(tree) or "?"
	local ok, pv = pcall(function() return tree:GetPivot().Position end)
	local r = getRoot()
	local text = (prefix and (prefix .. ": ") or "") .. class .. "  -  " .. #sectionsOf(tree) .. " sections"
	if ok and r then text = text .. "  -  " .. math.floor((pv - r.Position).Magnitude) .. "m away" end
	if LT.prevTree == tree then treePrev.Caption(text) return end
	LT.prevTree = tree
	treePrev.Model(tree, text)
end
local function refreshInfo()
	local c = LT.class
	if not c then
		axeRow.Text = "Best axe: -"
		countRow.Text = "Trees of this type: -"
		treePrev.Clear("Pick a tree type")
		LT.prevTree = nil
		return
	end
	local _, nm, dmg = bestAxe(c)
	axeRow.Text = nm and ("Best axe: " .. nm .. "  (damage " .. tostring(dmg) .. ")") or "Best axe: none found in your backpack"
	local n = 0
	for _, f in ipairs(regions()) do for _, m in ipairs(f:GetChildren()) do if m:IsA("Model") and classOf(m) == c and #sectionsOf(m) > 1 then n = n + 1 end end end
	countRow.Text = "Trees of this type: " .. n
	if not LT.busy then showTreePreview(pickTree(c, LT.pick)) end
end
local function syncType()
	if not typeDd then return end
	local shown = shownClasses()
	local opts = shown
	if #opts == 0 then opts = { (LT.classes and #LT.classes > 0) and "No match" or "None" } end
	typeDd.SetOptions(opts, true)
	local at = 1
	for i, c in ipairs(opts) do if c == LT.class then at = i end end
	typeDd.Set(at, true)
end
local function setClass(c)
	LT.class = c
	syncType()
	refreshInfo()
end
local function refreshTypes()
	local list = scanClasses()
	local keep = false
	for _, c in ipairs(list) do if c == LT.class then keep = true end end
	if not keep then setClass(list[1]) else refreshInfo() end
	notify("Found " .. #list .. " tree type(s)", T.Good)
end
local function applyTypeSearch(text)
	typeSearch = (text or ""):lower()
	local shown = shownClasses()
	local has = false
	for _, c in ipairs(shown) do if c == LT.class then has = true end end
	if not has and shown[1] then LT.class = shown[1] LT.prevTree = nil end
	syncType()
	refreshInfo()
end

	local c = shownClasses()[i]
	if c then LT.class = c LT.prevTree = nil refreshInfo() end

local VEHICLE_PRESETS = {
	Pickup1_Vehicle = { Size = Vector3.new(10, 6, 18), Offset = CFrame.new(0, 3, 11.5) },
	UtilityTruck_Vehicle = { Size = Vector3.new(7, 3, 10), Offset = CFrame.new(0, 1.75, 5.75) },
	UtilityTruck2_Vehicle = { Size = Vector3.new(8, 3, 11), Offset = CFrame.new(0, 1.75, 6) },
}
local function vehicleTp(veh, pos)
	local target = CFrame.new(pos + Vector3.new(0, 1, 0))
	local anch = {}
	for _, d in ipairs(veh:GetDescendants()) do
		if d:IsA("BasePart") then anch[d] = d.Anchored d.Anchored = true end
	end
	local nameV = veh:FindFirstChild("ItemName")
	local preset = nameV and nameV:IsA("StringValue") and VEHICLE_PRESETS[nameV.Value]
	if preset then
		local pivot = veh:GetPivot()
		local move = target * pivot:Inverse()
		local op = OverlapParams.new()
		op.FilterType = Enum.RaycastFilterType.Exclude
		op.FilterDescendantsInstances = { veh, getChar() }
		op.RespectCanCollide = false
		local done = {}
		for _, part in ipairs(Workspace:GetPartBoundsInBox(pivot * preset.Offset, preset.Size, op)) do
			if not part.Anchored then
				local m = part:FindFirstAncestorOfClass("Model")
				if m and m ~= Workspace then
					if not done[m] then done[m] = true m:PivotTo(move * m:GetPivot()) end
				elseif not done[part] then
					done[part] = true
					part.CFrame = move * part.CFrame
				end
			end
		end
	end
	veh:PivotTo(target)
	task.wait(0.05)
	for p, a in pairs(anch) do if p.Parent then p.Anchored = a end end
end
local function tpTo(v)
	local r, h = getRoot(), getHum()
	if not r then return end
	local seat = h and h.SeatPart
	local veh = seat and seat:FindFirstAncestorOfClass("Model")
	if veh and LT.carry ~= false and veh ~= getChar() and (veh:FindFirstChild("Configuration", true) or veh:FindFirstChild("ItemName")) then
		vehicleTp(veh, v)
		return
	end
	if h and h.SeatPart then h.Sit = false task.wait(0.05) end
	r.CFrame = CFrame.new(v + Vector3.new(0, 3, 0))
end

-- ---------------------------------------------------- WOOD: get trees (quantity) + chop logs into sections
local function cutAtHeight(section, tool, axeName, class, height, stop)
	local idObj = section:FindFirstChild("ID")
	if not idObj then return end
	local cutEvent = section:FindFirstChild("CutEvent") or section.Parent:FindFirstChild("CutEvent") or (section.Parent.Parent and section.Parent.Parent:FindFirstChild("CutEvent"))
	if not cutEvent then return end
	local args = {
		sectionId = idObj.Value, faceVector = Vector3.new(0, 0, -1), height = height,
		hitPoints = axeDamage(axeName, class), cooldown = 0, cuttingClass = "Axe", tool = tool,
	}
	for _ = 1, LT.fires do
		if not section.Parent or stop() then break end
		RemoteProxy:FireServer(cutEvent, args)
		task.wait(math.max(LT.fireDelay / 100, 0.005))
	end
end
local function woodCount(model)
	local n = 0
	for _, d in ipairs(model:GetDescendants()) do
		if d:IsA("BasePart") and d.Name == "WoodSection" then n = n + 1 end
	end
	return n
end
local function chopLogsIntoSections(alive)
	local lm = Workspace:FindFirstChild("LogModels")
	if not lm then notify("No LogModels folder", T.Warn) return end
	local queue = myLogs()
	if #queue == 0 then notify("None of your logs found", T.Warn) return end
	local cuts = 0
	while #queue > 0 and alive() do
		local model = table.remove(queue, 1)
		if model and model.Parent then
			local tc = model:FindFirstChild("TreeClass")
			local class = tc and tc.Value or "Generic"
			local tool, axeName = bestAxe(class)
			if not tool then notify("No axe in your backpack", T.Warn) return end
			local iw = model:FindFirstChild("InnerWood", true)
			local stump
			for _, d in ipairs(model:GetDescendants()) do
				if d:IsA("Weld") or d:IsA("ManualWeld") then
					local p0, p1 = d.Part0, d.Part1
					if iw and p0 == iw and p1 and p1.Name == "WoodSection" then stump = p1 end
					if iw and p1 == iw and p0 and p0.Name == "WoodSection" then stump = p0 end
				end
			end
			local guard = 0
			while model.Parent and woodCount(model) > 1 and alive() and guard < 200 do
				guard = guard + 1
				local conn = {}
				for _, d in ipairs(model:GetDescendants()) do
					if (d:IsA("Weld") or d:IsA("ManualWeld")) and d.Name == "Tree Weld" then
						local p0, p1 = d.Part0, d.Part1
						if p0 and p1 and p0.Name == "WoodSection" and p1.Name == "WoodSection" then
							conn[p0] = (conn[p0] or 0) + 1
							conn[p1] = (conn[p1] or 0) + 1
						end
					end
				end
				local tip
				for sec, n in pairs(conn) do
					if n == 1 and sec ~= stump and sec.Parent then tip = sec break end
				end
				if not tip then break end
				local weld, partner
				for _, d in ipairs(model:GetDescendants()) do
					if (d:IsA("Weld") or d:IsA("ManualWeld")) and d.Name == "Tree Weld" then
						local p0, p1 = d.Part0, d.Part1
						if p0 == tip and p1 and p1.Name == "WoodSection" then weld, partner = d, p1 break end
						if p1 == tip and p0 and p0.Name == "WoodSection" then weld, partner = d, p0 break end
					end
				end
				if not weld then break end
				local cutPart = partner:FindFirstChild("ID") and partner or tip
				local jointPos = (weld.Part0.CFrame * weld.C0).Position
				local localY = (cutPart.CFrame:Inverse() * CFrame.new(jointPos)).Position.Y
				local height = math.clamp(localY + cutPart.Size.Y / 2, 0.05, cutPart.Size.Y - 0.05)
				local r = getRoot()
				if r then
					r.CFrame = CFrame.new(jointPos + cutPart.CFrame.RightVector * 4)
					r.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
					task.wait(0.1)
				end
				local before = {}
				for _, m in ipairs(lm:GetChildren()) do before[m] = true end
				cutAtHeight(cutPart, tool, axeName, class, height, function() return not alive() end)
				cuts = cuts + 1
				task.wait(0.1)
				for _, m in ipairs(lm:GetChildren()) do
					if not before[m] and m:IsA("Model") and ownedByMe(m) and woodCount(m) > 1 then table.insert(queue, m) end
				end
				setStatus("Chopping logs - " .. cuts .. " cut(s), " .. #queue .. " queued")
			end
		end
	end
end

local function fetchNpcId(name, npc)

local function safeInvoke(chatted, arg, action)
	local done = false
	task.spawn(function()
		pcall(function() chatted:InvokeServer(arg, action) end)

	end)

	while not done and os.clock() - t < 4 do task.wait(0.03) end
	return done
end

-- boxes ("Box" models with BoxItemName + Main) live in Workspace.Stores.ShopItems folders; match a folder to a store by distance to its counter
local function shopItemsFor(e)
	local stores = Workspace:FindFirstChild("Stores")
	local cp = counterPart(e)
	if not (stores and cp) then return nil end
	local best, bestD
	for _, ch in ipairs(stores:GetChildren()) do
		if ch.Name == "ShopItems" then
			local sample = ch:FindFirstChildWhichIsA("BasePart", true)
			if sample then
				local d = (sample.Position - cp.Position).Magnitude
				if not bestD or d < bestD then best, bestD = ch, d end
			end
		end
	end
	return best
end
local function scanAllItems()
	SH.all = {}

			for _, box in ipairs(cont:GetChildren()) do
				local nv = box.Name == "Box" and box:FindFirstChild("BoxItemName")
				if nv and nv:IsA("StringValue") and not seen[nv.Value] then
					seen[nv.Value] = true
					local f = info and info:FindFirstChild(nv.Value)
					local n = f and f:FindFirstChild("ItemName")
					local p = f and f:FindFirstChild("Price")
					local im = f and f:FindFirstChild("ItemImage")
					table.insert(SH.all, {
						key = nv.Value, name = n and tostring(n.Value) or nv.Value,
						price = p and tonumber((tostring(p.Value):gsub("[^%d]", ""))) or nil,
						image = im and im.Value or nil, store = si,
					})
				end
			end
		end
	end
	table.sort(SH.all, function(a, b) return a.name < b.name end)
end

	if img ~= "" then
		if tonumber(img) then img = "rbxassetid://" .. img end

			total = total + 1
			if #SH.items < 40 then
				it.label = it.name .. (it.price and (" ($" .. commas(it.price) .. ")") or "") .. (only == 0 and ("  [" .. STORES[it.store][1] .. "]") or "")

		end
	end

-- flow (same order the Dynxe script uses): item onto the counter first, THEN stand at the cashier and get the ID, then confirm,
-- then bring the item back to dropCF. The cashier ID can only be fetched while you are standing near the cashier.
local function buyBox(e, box, alive, dropCF)
	local main = box:FindFirstChild("Main")
	local npc = npcOf(e)
	local _, chatted = dialogRemotes()
	local cp = counterPart(e)
	if not (main and npc and chatted and cp) then notify("Store parts not found", T.Bad) return false end
	-- 1. go to the item, take ownership, put it on the counter
	local r = getRoot()
	if r then r.CFrame = CFrame.new(main.Position + Vector3.new(0, 4, 0)) task.wait(0.15) end
	claimDrag(box, 2)
	if main.Parent then
		local up = cp.CFrame.UpVector
		main.CFrame = CFrame.new(cp.Position + up * (cp.Size.Y / 2 + main.Size.Y / 2 + 0.05))
		main.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
		main.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
	end
	task.wait(0.1)
	-- 2. stand at the cashier, then get the dialog ID
	r = getRoot()
	if r then r.CFrame = e[4] task.wait(0.4) end
	if not npc:FindFirstChild("Dialog") then Instance.new("Dialog", npc) end
	local id = SH.npcIds[e[3]] or fetchNpcId(e[3], npc)
	if not id then notify("Could not get the cashier's ID - try again", T.Bad) return false end
	local arg = { ID = id, Character = npc, Name = e[3], Dialog = npc.Dialog }
	-- 3. confirm until the item belongs to you
	local deadline = os.clock() + 30
	local bought = false
	while alive() and os.clock() < deadline do
		local st = itemState(main)
		if st == "success" then bought = true break end
		if st == "gone" then break end
		if not safeInvoke(chatted, arg, "Initiate") then
			-- cashier did not answer: clear the dialog and fetch a fresh ID
			task.wait(0.6)
			for _ = 1, 3 do safeInvoke(chatted, arg, "EndChat") task.wait(0.3) end
			SH.npcIds[e[3]] = nil
			local nid = fetchNpcId(e[3], npc)
			if nid then arg.ID = nid end
		else
			local inner = os.clock() + 3
			while alive() and os.clock() < inner do
				st = itemState(main)
				if st ~= "pending" then break end
				if not safeInvoke(chatted, arg, "ConfirmPurchase") then break end
				task.wait(0.1)
			end
			st = itemState(main)
			if st == "success" then bought = true end
			safeInvoke(chatted, arg, "EndChat")
			if bought or st == "gone" then break end
			task.wait(0.2)
		end
	end
	safeInvoke(chatted, arg, "EndChat")
	-- 4. bring the item back to where you started
	if bought and dropCF and main.Parent then
		claimDrag(main:FindFirstAncestorOfClass("Model") or box, 1)
		if main.Parent then
			main.CFrame = dropCF
			main.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
		end
	end
	return bought
end

local SEL = { list = {}, hl = {}, click = nil, same = false, keep = true, back = false, timeout = 3, X = 4, Y = 1, Z = 4, pad = 2 }
local function setSel(model, on)
	if on and not SEL.list[model] then
		SEL.list[model] = true
		SEL.hl[model] = new("Highlight", { Adornee = model, FillColor = T.Accent, FillTransparency = 0.5, OutlineColor = Color3.new(1, 1, 1) }, EspGui)
	elseif not on and SEL.list[model] then
		SEL.list[model] = nil
		if SEL.hl[model] then SEL.hl[model]:Destroy() SEL.hl[model] = nil end
	end
end
local function clearSel()
	for m in pairs(SEL.list) do setSel(m, false) end
end
local function selList()
	local out = {}
	for m in pairs(SEL.list) do
		if m.Parent then table.insert(out, m) else setSel(m, false) end
	end
	return out
end
local selRow
local function updSel() if selRow then selRow.Text = "Selected: " .. #selList() .. " object(s)" end end
local function moveModel(model, goal)
	local part = mainPartOf(model)
	if not (part and part.Parent) then return false end
	local r = getRoot()
	if r then
		local flat = (r.Position - part.Position) * Vector3.new(1, 0, 1)
		if flat.Magnitude > 10 then r.CFrame = CFrame.new(part.Position + Vector3.new(0, 3, 0)) task.wait(0.05) end
	end
	claimDrag(model, SEL.timeout)
	if part.Parent then part.CFrame = goal end
	task.wait(0.05)
	return true
end
local function sendSelection(centerCF, grid, alive)
	local list = selList()
	if #list == 0 then notify("Nothing selected", T.Warn) return end
	local r = getRoot()
	local home = r and r.CFrame
	local cell, cellY = 4, 2
	if grid then
		for _, m in ipairs(list) do
			local ok, sz = pcall(function() return m:GetExtentsSize() end)
			if ok then cell = math.max(cell, sz.X, sz.Z) cellY = math.max(cellY, sz.Y) end
		end
		cell, cellY = cell + SEL.pad, cellY + SEL.pad * 0.5
	end
	for i, m in ipairs(list) do
		if not alive() then break end
		local k = i - 1
		local col, row, layer
		if grid then
			col, row, layer = k % SEL.X, math.floor(k / SEL.X) % SEL.Z, math.floor(k / (SEL.X * SEL.Z))
		else
			col, row, layer = k % 5, math.floor(k / 5), 0
		end
		local step = grid and cell or 5
		local goal = centerCF * CFrame.new(col * step, 1 + layer * (grid and cellY or 3), -row * step)
		moveModel(m, goal)
	end
	if SEL.back and home then local rr = getRoot() if rr then rr.CFrame = home end end
	if not SEL.keep then clearSel() end
	updSel()
end

local function findVehConfig()
	local h = getHum()
	local seat = h and h.SeatPart
	if not seat then return nil, nil end
	local v = seat
	while v.Parent and v.Parent ~= Workspace and v.Parent.Name ~= "PlayerModels" do v = v.Parent end
	if v and v:IsA("Model") then
		local c = v:FindFirstChild("Configuration", true)
		if c and c:FindFirstChild("MaxSpeed") then return c, v end
	end
	return nil, nil
end
local function readVeh(cfg)
	local out = {}
	for _, n in ipairs({ "MaxSpeed", "SteerAngle", "SteerVelocity" }) do
		local s = cfg:FindFirstChild(n)
		out[n] = s and s:IsA("ValueBase") and s.Value or nil
	end
	return out
end
local function writeVeh(cfg, vals)
	if not (cfg and vals) then return end
	for n, v in pairs(vals) do
		local s = cfg:FindFirstChild(n)
		if s and s:IsA("ValueBase") then s.Value = v end
	end
end
local function applyVeh(name, value)
	VEH.custom[name] = value
	if VEH.cfg then
		local s = VEH.cfg:FindFirstChild(name)
		if s and s:IsA("ValueBase") then s.Value = value end
	else
		notify("Sit in a vehicle first", T.Warn)
	end
end
local function padFire()
	local i = ReplicatedStorage:FindFirstChild("Interaction")
	local rp = i and i:FindFirstChild("RemoteProxy")
	if VEH.pad and rp then rp:FireServer(VEH.pad) end
end

local function spawnAndGetColor()
	local watch = Workspace:FindFirstChild("PlayerModels") or Workspace
	local existing = {}
	for _, m in ipairs(watch:GetChildren()) do existing[m] = true end
	local done, result = false, nil
	local conn = watch.ChildAdded:Connect(function(model)
		if done or existing[model] or not model:IsA("Model") then return end
		local st = model:WaitForChild("Settings", 0.5)
		local cv = st and st:WaitForChild("Color", 0.5)
		if not cv then return end
		local dl = os.clock() + 0.4
		while cv.Value == 0 and os.clock() < dl do task.wait(0.02) end
		if cv.Value ~= 0 and not done then done = true result = cv.Value end
	end)
	padFire()
	local dl = os.clock() + 0.5
	while not done and os.clock() < dl and VEH.roll do task.wait(0.02) end
	conn:Disconnect()
	return result
end

	while Live.Running do

		if seat ~= lastSeat then
			lastSeat = seat
			if seat then
				VEH.cfg, VEH.veh = nil, nil
				for _ = 1, 6 do
					local c, v = findVehConfig()
					if c then VEH.cfg, VEH.veh = c, v break end
					task.wait(0.5)
				end
				if VEH.cfg then
					if not VEH.orig[VEH.veh] then VEH.orig[VEH.veh] = readVeh(VEH.cfg) end
					VEH.defaults = readVeh(VEH.cfg)
					vehRow.Text = "Vehicle: " .. VEH.veh.Name

local WD = { water = {}, boulders = {}, bridge = nil, volcano = true, scanned = false, enhanced = nil, tundraConn = nil }
local function scanWorld()
	if WD.scanned then return end
	WD.scanned = true
	local wr = Workspace:FindFirstChild("Water")
	if wr then
		for i, o in ipairs(wr:GetDescendants()) do
			if i % 500 == 0 then task.wait() end
			if o:IsA("BasePart") and o.Name == "Water" then table.insert(WD.water, { p = o, t = o.Transparency }) end
		end
	end
	local snow = Workspace:FindFirstChild("Region_Snow")
	local sp = snow and snow:FindFirstChild("PartSpawner")
	if sp then
		for i, o in ipairs(sp:GetDescendants()) do
			if i % 500 == 0 then task.wait() end
			if o:IsA("BasePart") and (o.Name == "Boulder" or o.Name == "SmallBoulder") and not (o:FindFirstChild("LavaLight") or o:FindFirstChild("Fire")) then
				table.insert(WD.boulders, { p = o, t = o.Transparency })
			end
		end
	end
end
local function sweepVolcano()
	local r = Workspace:FindFirstChild("Region_Volcano")
	local sp = r and r:FindFirstChild("PartSpawner")
	if not sp then return end
	for _, o in ipairs(sp:GetDescendants()) do
		if o.Name == "VolcanoBoulder" then pcall(function() o:Destroy() end) end
	end
end

		end

-- ---------------------------------------------------- PROTECTION: axe recovery
local AR = { on = false, conn = nil, diedConn = nil, deathPos = nil, radius = 50, maxAxes = 9, timeout = 3 }
local function heldToolCount()
	local n = 0
	for _, c in ipairs({ LocalPlayer:FindFirstChildOfClass("Backpack"), getChar(), Workspace:FindFirstChild(LocalPlayer.Name) }) do
		if c then for _, v in ipairs(c:GetChildren()) do if v.Name == "Tool" and v:IsA("Tool") then n = n + 1 end end end
	end
	return n
end
local function recoverAxes()
	task.wait(0.1)
	if not AR.on then return end
	local interaction = ReplicatedStorage:FindFirstChild("Interaction")
	local ci = interaction and interaction:FindFirstChild("ClientInteracted")
	local pm = Workspace:FindFirstChild("PlayerModels")
	if not (ci and pm) then return end
	local axes = {}
	for _, o in ipairs(pm:GetDescendants()) do
		if o.Name == "Model" and o:IsA("Model") then
			local of = o:FindFirstChild("Owner")
			local os2 = of and of:FindFirstChild("OwnerString")
			if os2 and os2.Value == LocalPlayer.Name then
				local handle = o:FindFirstChild("Handle") or o.PrimaryPart
				if handle and (not AR.deathPos or (handle.Position - AR.deathPos).Magnitude <= AR.radius) then table.insert(axes, o) end
			end
		end
		if #axes >= AR.maxAxes then break end
	end
	local got = 0
	for _, axe in ipairs(axes) do
		local handle = axe:FindFirstChild("Handle") or axe.PrimaryPart
		if handle then
			local before = heldToolCount()
			local dl = os.clock() + AR.timeout
			while os.clock() < dl and axe.Parent do
				ci:FireServer(axe, "Pick up tool", handle.CFrame)
				task.wait(0.15)
				if heldToolCount() > before then got = got + 1 break end
			end
		end
	end
	if got > 0 then notify("Recovered " .. got .. " axe(s)", T.Good) end
end
local function hookDeath(char)
	if AR.diedConn then AR.diedConn:Disconnect() end
	local h = char:WaitForChild("Humanoid", 10)
	if not h then return end
	AR.diedConn = h.Died:Connect(function()
		local r = char:FindFirstChild("HumanoidRootPart")
		AR.deathPos = r and r.Position or nil
	end)
end
local function setAxeRecovery(v)
	AR.on = v
	if AR.conn then AR.conn:Disconnect() AR.conn = nil end
	if AR.diedConn then AR.diedConn:Disconnect() AR.diedConn = nil end
	if not v then return end
	if LocalPlayer.Character then task.spawn(hookDeath, LocalPlayer.Character) end
	AR.conn = LocalPlayer.CharacterAdded:Connect(function(char)
		task.spawn(hookDeath, char)
		task.spawn(recoverAxes)
	end)
end
local Pr = Section(ProtTab, "Axe Recovery", "After you die, picks your axes back up from where you died", "godmode")
Toggle(Pr, "Axe Recovery", true, setAxeRecovery, nil, "lt2axerecover")
setAxeRecovery(true)
local PrOpt = Collapse(Pr, "Recovery Settings")
Slider(PrOpt, "Search Radius (studs)", 10, 200, 50, function(v) AR.radius = v end)
Slider(PrOpt, "Max Axes Per Respawn", 1, 20, 9, function(v) AR.maxAxes = v end)
Slider(PrOpt, "Pickup Timeout (s)", 1, 10, 3, function(v) AR.timeout = v end)
addCleanup(function() setAxeRecovery(false) end)

-- Tree ESP: billboards on the closest trees only, rebuilt every 1.5s
local espGuis = {}
local function clearEsp() for m, g in pairs(espGuis) do g:Destroy() espGuis[m] = nil end end
task.spawn(function()
	while Live.Running do
		task.wait(1.5)
		if not LT.esp then
			if next(espGuis) then clearEsp() end
		else
			local r = getRoot()
			local list = {}
			if r then
				for _, f in ipairs(regions()) do
					for _, m in ipairs(f:GetChildren()) do
						local c = m:IsA("Model") and classOf(m)
						if c and (LT.espAll or c == LT.class) then
							local b = baseSection(m)
							if b then
								local d = (b.Position - r.Position).Magnitude
								if d <= LT.espMax then table.insert(list, { m = m, b = b, d = d, c = c }) end
							end
						end
					end
				end
			end
			table.sort(list, function(a, b) return a.d < b.d end)
			local want = {}
			for i = 1, math.min(#list, 60) do
				local e = list[i]
				want[e.m] = true
				local g = espGuis[e.m]
				if not g then
					g = new("BillboardGui", { Adornee = e.b, AlwaysOnTop = true, Size = UDim2.fromOffset(130, 18), StudsOffset = Vector3.new(0, 4, 0) }, EspGui)
					new("TextLabel", { Name = "L", Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, TextColor3 = T.Good, TextStrokeTransparency = 0.4, Font = Enum.Font.GothamBold, TextSize = 12 }, g)
					espGuis[e.m] = g
				end
				g.L.Text = e.c .. " (" .. #sectionsOf(e.m) .. ")  " .. math.floor(e.d) .. "m"
			end
			for m, g in pairs(espGuis) do if not want[m] then g:Destroy() espGuis[m] = nil end end
		end
	end
end)
