-- ---------------------------------------------------- helpers
local function rootOf(p) return p and p.Character and p.Character:FindFirstChild("HumanoidRootPart") end
local function alive(p)
	local h = p and p.Character and p.Character:FindFirstChildOfClass("Humanoid")
	return h ~= nil and h.Health > 0
end
local function hasTool(p, name)
	local ch, bp = p.Character, p:FindFirstChildOfClass("Backpack")
	return (ch and ch:FindFirstChild(name)) or (bp and bp:FindFirstChild(name))
end
local function roleOf(p)
	if hasTool(p, "Knife") then return "Murderer" end
	if hasTool(p, "Gun") then return "Sheriff" end
	return "Innocent"
end
local function findRole(role)
	for _, p in ipairs(Players:GetPlayers()) do
		if alive(p) and roleOf(p) == role and hasTool(p, role == "Murderer" and "Knife" or "Gun") then return p end
	end
	return nil
end
local function gunDrop()
	local g = Workspace:FindFirstChild("GunDrop", true)
	return g and g:IsA("BasePart") and g or nil
end
local function toolNamed(n)
	local ch, bp = LocalPlayer.Character, LocalPlayer:FindFirstChildOfClass("Backpack")
	local t = (ch and ch:FindFirstChild(n)) or (bp and bp:FindFirstChild(n))
	return t and t:IsA("Tool") and t or nil
end
local function equip(tool)
	if tool.Parent ~= LocalPlayer.Character then
		local h = getHum()
		if h then h:EquipTool(tool) else tool.Parent = LocalPlayer.Character end
	end
end
local lastWarn = 0
local function warn2(text) if os.clock() - lastWarn > 2 then lastWarn = os.clock() notify(text, T.Warn) end end
local function stopAll() M.token = M.token + 1 M.busy = false end
local function runSeq(fn)
	if M.busy then notify("Already running - press Stop first", T.Warn) return end
	M.token = M.token + 1
	local my = M.token
	M.busy = true
	task.spawn(function()
		local good, err = pcall(fn, function() return Live.Running and M.token == my end)
		if not good then warn2("Error: " .. tostring(err):sub(1, 60)) end
		if M.token == my then M.busy = false end
	end)
end

-- ---------------------------------------------------- role ESP (highlight + name tag)
local objs = {}
local function clearEsp()
	for p, o in pairs(objs) do
		if o.hl then o.hl:Destroy() end
		if o.bb then o.bb:Destroy() end
		objs[p] = nil
	end
end
local gunHl, gunBb
local function clearGun()
	if gunHl then gunHl:Destroy() gunHl = nil end
	if gunBb then gunBb:Destroy() gunBb = nil end
end
local function updateEsp()
	local myRoot = getRoot()
	for _, p in ipairs(Players:GetPlayers()) do
		local o = objs[p]
		local show = M.esp and p ~= LocalPlayer and alive(p) and p.Character
		local role = show and roleOf(p)
		if show and role == "Innocent" and not M.showInno then show = false end
		if not show then
			if o then o.hl.Enabled = false o.bb.Enabled = false end
		else
			if not o then
				o = {}
				o.hl = new("Highlight", { FillTransparency = M.fill, OutlineTransparency = 0, DepthMode = Enum.HighlightDepthMode.AlwaysOnTop }, EspGui)
				o.bb = new("BillboardGui", { Size = UDim2.fromOffset(160, 36), StudsOffset = Vector3.new(0, 3.2, 0), AlwaysOnTop = true }, EspGui)
				o.lbl = new("TextLabel", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, TextSize = 13, TextStrokeTransparency = 0.4 }, o.bb)
				objs[p] = o
			end
			local col = COLORS[role]
			o.hl.Adornee = p.Character
			o.hl.FillColor, o.hl.OutlineColor, o.hl.FillTransparency, o.hl.Enabled = col, col, M.fill, true
			local head = p.Character:FindFirstChild("Head") or rootOf(p)
			o.bb.Adornee = head
			o.bb.Enabled = M.names
			local dist = (myRoot and rootOf(p)) and math.floor((rootOf(p).Position - myRoot.Position).Magnitude) or 0
			o.lbl.Text = (role == "Innocent" and p.DisplayName or (role .. " - " .. p.DisplayName)) .. "  [" .. dist .. "]"
			o.lbl.TextColor3 = col
		end
	end
	for p, o in pairs(objs) do
		if not p.Parent then o.hl:Destroy() o.bb:Destroy() objs[p] = nil end
	end
	-- dropped gun
	local g = M.gunEsp and gunDrop()
	if g then
		if not gunHl then
			gunHl = new("Highlight", { FillColor = Color3.fromRGB(255, 220, 60), OutlineColor = Color3.fromRGB(255, 220, 60), FillTransparency = 0.3, DepthMode = Enum.HighlightDepthMode.AlwaysOnTop }, EspGui)
			gunBb = new("BillboardGui", { Size = UDim2.fromOffset(120, 24), AlwaysOnTop = true, StudsOffset = Vector3.new(0, 2, 0) }, EspGui)
			new("TextLabel", { Name = "T", Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, TextSize = 13, TextColor3 = Color3.fromRGB(255, 220, 60), TextStrokeTransparency = 0.4 }, gunBb)
		end
		gunHl.Adornee, gunBb.Adornee = g, g
		local d = myRoot and math.floor((g.Position - myRoot.Position).Magnitude) or 0
		gunBb.T.Text = "DROPPED GUN [" .. d .. "]"
	else
		clearGun()
	end
end
onUnload(function() clearEsp() clearGun() end)

-- ---------------------------------------------------- gun grab
local function grabGun(ok)
	local g, r = gunDrop(), getRoot()
	if not g or not r then notify("No dropped gun right now", T.Warn) return false end
	local home = r.CFrame
	r.CFrame = g.CFrame + Vector3.new(0, 1.5, 0)
	r.AssemblyLinearVelocity = Vector3.zero
	local t0 = os.clock()
	while (not ok or ok()) and gunDrop() and os.clock() - t0 < 1 do task.wait(0.05) end
	local root = getRoot()
	if root then root.CFrame = home root.AssemblyLinearVelocity = Vector3.zero end
	return true
end
task.spawn(function()
	while Live.Running do
		if M.autoGun and not M.busy and gunDrop() and roleOf(LocalPlayer) ~= "Sheriff" and alive(LocalPlayer) then
			grabGun()
			task.wait(2)
		end
		task.wait(0.3)
	end
end)

-- ---------------------------------------------------- coins
local function coins()
	local list, r = {}, getRoot()
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("BasePart") and d.Name == "Coin_Server" and d.Parent then table.insert(list, d) end
	end
	if r then table.sort(list, function(a, b) return (a.Position - r.Position).Magnitude < (b.Position - r.Position).Magnitude end) end
	return list
end
local function collectCoins(ok)
	local list = coins()
	if #list == 0 then notify("No coins found", T.Warn) return end
	local r = getRoot()
	local home = r and r.CFrame
	local n = 0
	for _, c in ipairs(list) do
		if ok and not ok() then break end
		local root = getRoot()
		if not root then break end
		if c.Parent then
			root.CFrame = c.CFrame
			root.AssemblyLinearVelocity = Vector3.zero
			n = n + 1
			task.wait(M.coinDelay)
		end
	end
	local root = getRoot()
	if root and home then root.CFrame = home end
	notify("Visited " .. n .. " coins", T.Good)
end
task.spawn(function()
	while Live.Running do
		if M.coinFarm and not M.busy and alive(LocalPlayer) then
			local list = coins()
			local r = getRoot()
			if #list > 0 and r then
				local c = list[1]
				r.CFrame = c.CFrame
				r.AssemblyLinearVelocity = Vector3.zero
			end
		end
		task.wait(math.max(M.coinDelay, 0.05))
	end
end)

-- ---------------------------------------------------- target tracker + aim solver (auto shoot AND fling use this)
-- Other players are drawn a little in the past (network + interpolation), but the server judges shots, and the physics
-- contact that flings someone, against where they REALLY are. So every target gets a short history of the positions we
-- see, and solve() predicts where they will be when the shot / contact lands:
--   * horizontal: a fast fit (last ~0.14 s) blended with a slow fit (last ~0.45 s). A runner is extrapolated; someone
--     spamming A/D is aimed at the middle of their wobble instead of a point they are about to leave.
--   * vertical: a real jump arc (gravity) that stops at the floor, so a jumping target is aimed at where they will be
--     in the air, not where they were.
--   * confidence 0..1: low for erratic strafing, for a jump spammer standing on the ground (they can take off at any
--     moment) and for a target that will land before the shot does. The shooter waits for a good moment.
local TRK = {}
local rpFloor = RaycastParams.new()
rpFloor.FilterType = Enum.RaycastFilterType.Exclude
local StatsSvc = game:GetService("Stats")
local function pingTime() -- round trip in seconds
	local rtt
	pcall(function() rtt = StatsSvc.Network.ServerStatsItem["Data Ping"]:GetValue() / 1000 end)
	if not rtt or rtt <= 0 then
		local one = 0.06
		pcall(function() one = LocalPlayer:GetNetworkPing() end)
		rtt = one * 2
	end
	return math.clamp(rtt, 0.02, 1)
end
-- how far behind the server the positions we see are: our ping (x tuning) + the interpolation delay
local function lagTime()
	return math.clamp(pingTime() * M.lagMul + M.interp, 0, 0.9)
end
local function trackerOf(p)
	local k = TRK[p]
	if not k then
		k = { s = {}, jumps = {}, lastVy = 0, floorY = nil, floorT = 0 }
		TRK[p] = k
	end
	k.want = os.clock()
	return k
end
bind(RunService.Heartbeat, function()
	local now = os.clock()
	for p, k in pairs(TRK) do
		if not p.Parent or now - k.want > 2 then
			TRK[p] = nil
		else
			local r = rootOf(p)
			if r then
				local s = k.s
				local last = s[#s]
				if last and (r.Position - last.pos).Magnitude > 40 then table.clear(s) end -- respawn / teleport: start over
				table.insert(s, { t = now, pos = r.Position })
				while #s > 0 and now - s[1].t > 0.6 do table.remove(s, 1) end
			end
		end
	end
end)
-- least-squares line through the newest `w` seconds of samples: velocity + fitted position at the newest sample
local function lsq(s, w)
	local n = #s
	if n < 3 then return nil end
	local tn = s[n].t
	local i0 = n
	while i0 > 1 and tn - s[i0 - 1].t <= w do i0 = i0 - 1 end
	local cnt = n - i0 + 1
	if cnt < 3 then return nil end
	local st, stt, sp, stp = 0, 0, Vector3.zero, Vector3.zero
	for i = i0, n do
		local d = s[i]
		local tt = d.t - tn
		st = st + tt
		stt = stt + tt * tt
		sp = sp + d.pos
		stp = stp + d.pos * tt
	end
	local den = cnt * stt - st * st
	if math.abs(den) < 1e-9 then return nil end
	local v = (stp * cnt - sp * st) / den
	return v, (sp - v * st) / cnt
end
local function floorOf(k, r)
	local now = os.clock()
	if now - k.floorT > 0.12 then
		k.floorT = now
		local ex = {}
		for _, pl in ipairs(Players:GetPlayers()) do if pl.Character then table.insert(ex, pl.Character) end end
		rpFloor.FilterDescendantsInstances = ex
		local res = Workspace:Raycast(r.Position, Vector3.new(0, -60, 0), rpFloor)
		k.floorY = res and res.Position.Y or nil
	end
	return k.floorY
end
-- where will `p` be `horizon` seconds from now? returns position, confidence (0..1), short description
local function solve(p, horizon)
	local r = rootOf(p)
	if not r then return nil end
	local k = trackerOf(p)
	local s = k.s
	local vS, aS = lsq(s, 0.14)
	local vL = lsq(s, 0.45)
	if not (vS and vL) then
		local v = r.AssemblyLinearVelocity -- no history yet: replicated velocity, low confidence
		return r.Position + Vector3.new(v.X, 0, v.Z) * horizon, 0.25, "warming up"
	end
	-- horizontal: steady runner = trust the fast fit, wobbling strafer = trust the slow (averaged) fit
	local hS, hL = Vector3.new(vS.X, 0, vS.Z), Vector3.new(vL.X, 0, vL.Z)
	local steadyH = math.clamp(1 - (hS - hL).Magnitude / (hL.Magnitude + 10), 0, 1)
	local hv = hL:Lerp(hS, steadyH)
	if hv.Magnitude > 60 then hv = hv.Unit * 60 end
	local x, z = aS.X + hv.X * horizon, aS.Z + hv.Z * horizon
	-- vertical: jump arc with the floor as a stop
	local g = Workspace.Gravity
	local fy = floorOf(k, r)
	local dist = fy and (aS.Y - fy) or nil
	if dist and math.abs(vS.Y) < 3 and dist < 8 then k.stand = math.min(k.stand or dist, dist) end -- learn their standing height
	local stand = k.stand or 3.1
	local airborne
	if dist then airborne = dist > stand + 0.9 else airborne = math.abs(vS.Y) > 8 end
	local now = os.clock()
	if vS.Y > 28 and k.lastVy <= 28 then table.insert(k.jumps, now) end -- a takeoff
	k.lastVy = vS.Y
	while #k.jumps > 0 and now - k.jumps[1] > 2 do table.remove(k.jumps, 1) end
	local y, conf, state = aS.Y, steadyH, "grounded"
	if M.ballistic and (airborne or vS.Y > 6) then
		y = aS.Y + vS.Y * horizon - 0.5 * g * horizon * horizon
		local tLand = math.huge
		if fy then
			local floorCenter = fy + stand
			local disc = vS.Y * vS.Y + 2 * g * (aS.Y - floorCenter)
			if disc > 0 then tLand = (vS.Y + math.sqrt(disc)) / g end
			if y < floorCenter then y = floorCenter end
		end
		if tLand - horizon < 0.06 then conf, state = conf * 0.3, "landing" else conf, state = conf * 0.95, "airborne" end
	elseif #k.jumps >= 2 then
		conf, state = conf * 0.3, "jump spam" -- on the ground but about to take off again: wait for them to be in the air
	elseif steadyH < 0.5 then
		state = "strafing"
	end
	return Vector3.new(x, y, z), conf, state
end

-- one fling attempt. Sits where the target will REALLY be and sweeps a volume around that point; the sweep grows when
-- the target is hard to predict (jump spam / strafing), so they still get touched.
local function flingOnce(p, ok)
	local r, tr = getRoot(), rootOf(p)
	if not r or not tr then return false end
	local home, t0, p0, phase, flung = r.CFrame, os.clock(), tr.Position, 0, false
	trackerOf(p)
	local conn
	conn = RunService.Heartbeat:Connect(function(dt)
		local root, troot = getRoot(), rootOf(p)
		if not ok() or not root or not troot or os.clock() - t0 > (M.flingTime or 4) then conn:Disconnect() return end
		-- target flew away = success
		if (troot.Position - p0).Magnitude > 120 or troot.AssemblyLinearVelocity.Magnitude > 300 then flung = true conn:Disconnect() return end
		local pos, conf = solve(p, lagTime() + M.flingLead)
		if not pos then conn:Disconnect() return end
		phase = phase + dt
		local loose, sc = 1 - conf, M.flingSweep
		local off = Vector3.new(
			math.sin(phase * 37) * (1.2 + 2.2 * loose) * sc,
			math.sin(phase * 23 + 1.3) * (2.2 + 3.5 * loose) * sc + loose * 2,
			math.cos(phase * 31 + 2.1) * (1.2 + 2.2 * loose) * sc)
		root.CFrame = CFrame.new(pos + off)
		root.AssemblyLinearVelocity = Vector3.new(9e5, 9e5, 9e5)
		root.AssemblyAngularVelocity = Vector3.new(9e5, 9e5, 9e5)
	end)
	while conn.Connected do task.wait(0.05) end
	local root = getRoot()
	if root then
		root.AssemblyLinearVelocity, root.AssemblyAngularVelocity = Vector3.zero, Vector3.zero
		root.CFrame = home
	end
	return flung
end
local function flingPlayer(p)
	local r, tr = getRoot(), rootOf(p)
	if not r or not tr then notify("Target has no character", T.Warn) return end
	runSeq(function(ok)
		local flung = false
		for attempt = 1, (M.flingRetries or 1) + 1 do
			if not ok() then break end
			flung = flingOnce(p, ok)
			if flung then break end
			task.wait(0.15)
		end
		if not M.loopFling then notify(flung and "Flung them" or "Fling timed out", flung and T.Good or T.Warn) end
	end)
end

-- ---------------------------------------------------- murderer tools
local function stab(p, ok, maxTime)
	local t0 = os.clock()
	while ok() and alive(p) and os.clock() - t0 < (maxTime or 2) do
		local r, tr = getRoot(), rootOf(p)
		local knife = toolNamed("Knife")
		if not r or not tr or not knife then return false end
		equip(knife)
		local back = tr.CFrame * CFrame.new(0, 0, 2.5)
		r.CFrame = CFrame.lookAt(back.Position, tr.Position)
		r.AssemblyLinearVelocity = Vector3.zero
		pcall(function() knife:Activate() end)
		task.wait(0.04)
	end
	return not alive(p)
end
local function killAll()
	if not toolNamed("Knife") then notify("You need the Knife (be the Murderer)", T.Warn) return end
	runSeq(function(ok)
		local r = getRoot()
		if not r then return end
		local home, n = r.CFrame, 0
		for _, p in ipairs(Players:GetPlayers()) do
			if not ok() then break end
			if p ~= LocalPlayer and alive(p) and rootOf(p) and stab(p, ok) then n = n + 1 end
		end
		local root = getRoot()
		if root then root.CFrame = home end
		notify("Eliminated " .. n, n > 0 and T.Good or T.Warn)
	end)
end
task.spawn(function()
	while Live.Running do
		if M.aura and not M.busy then
			local knife, r = toolNamed("Knife"), getRoot()
			if knife and r then
				for _, p in ipairs(Players:GetPlayers()) do
					local pr = p ~= LocalPlayer and alive(p) and rootOf(p)
					if pr and (pr.Position - r.Position).Magnitude <= M.auraR then
						equip(knife)
						pcall(function() knife:Activate() end)
						break
					end
				end
			end
		end
		task.wait(0.05)
	end
end)

-- escape helpers
local function groundAt(pos)
	rpGround.FilterDescendantsInstances = { getChar() or Workspace }
	local res = Workspace:Raycast(pos + Vector3.new(0, 150, 0), Vector3.new(0, -400, 0), rpGround)
	return res and (res.Position + Vector3.new(0, 4, 0)) or (pos + Vector3.new(0, 6, 0))
end
local function awayFromMurderer(dist)
	local m, r = findRole("Murderer"), getRoot()
	local mr = rootOf(m)
	if not (mr and r) then return false end
	local dir = r.Position - mr.Position
	dir = Vector3.new(dir.X, 0, dir.Z)
	dir = dir.Magnitude > 1 and dir.Unit or Vector3.new(1, 0, 0)
	r.CFrame = CFrame.new(groundAt(r.Position + dir * dist))
	r.AssemblyLinearVelocity = Vector3.zero
	return true
end

end)

-- smooth (tweened) coin farm
local visited, coinCache, coinCacheT = setmetatable({}, { __mode = "k" }), {}, 0
local function nextCoin(from)
	if os.clock() - coinCacheT > 1 then coinCache, coinCacheT = coins(), os.clock() end
	local best, bd
	for _, c in ipairs(coinCache) do
		local seen = visited[c]
		if c.Parent and not (seen and os.clock() - seen < 3) then
			local d = (c.Position - from).Magnitude
			if not bd or d < bd then best, bd = c, d end
		end
	end
	return best, bd
end
task.spawn(function()
	while Live.Running do
		local moved = false
		if M.safeFarm and not M.coinFarm and not M.busy and alive(LocalPlayer) then
			local r = getRoot()
			local c, d
			if r then c, d = nextCoin(r.Position) end
			if c then
				local t = math.max(d / math.max(M.tweenSpd, 10), 0.03)
				local tw = TweenService:Create(r, TweenInfo.new(t, Enum.EasingStyle.Linear), { CFrame = c.CFrame })
				tw:Play()
				task.wait(t)
				visited[c] = os.clock() -- move on to the next coin right away
				moved = true
			elseif r then
				coinCacheT = 0
			end
		end
		if not moved then task.wait(0.2) end
	end
end)

local rpLos = RaycastParams.new()
rpLos.FilterType = Enum.RaycastFilterType.Exclude
local function lineOfSight(mr)
	local r = getRoot()
	if not r then return false end
	rpLos.FilterDescendantsInstances = { getChar() or Workspace }
	local res = Workspace:Raycast(r.Position, mr.Position - r.Position, rpLos)
	return res == nil or res.Instance:IsDescendantOf(mr.Parent)
end
local function fireAt(gun, pos, conf, state)
	local ok = pcall(function() gun.KnifeLocal.CreateBeam.RemoteFunction:InvokeServer(1, pos, "AH2") end)
	if not ok then
		local c = cam()
		if c then c.CFrame = CFrame.lookAt(c.CFrame.Position, pos) end
		pcall(function() gun:Activate() end)
	end
	if M.shotRow then M.shotRow.Set(math.floor(conf * 100 + 0.5) .. "% - " .. state, conf >= 0.6 and T.Good or T.Warn) end
end
-- equip and give the tool a moment to be ready: a shot fired the instant it is equipped is often ignored
local function equipGun(gun)
	if gun.Parent == LocalPlayer.Character then return true end
	equip(gun)
	local t0 = os.clock()
	while os.clock() - t0 < 0.6 and gun.Parent ~= LocalPlayer.Character do task.wait(0.03) end
	task.wait(0.12)
	return gun.Parent == LocalPlayer.Character
end
-- take ONE shot at a good moment: waits for enough history, then for the solver to be confident (or for the patience
-- timeout). strict = automatic mode: give up if they leave range / cover while we wait.
local function takeShot(m, strict)
	local gun = toolNamed("Gun")
	if not (gun and m and rootOf(m)) then return false end
	if not equipGun(gun) then return false end
	local t0 = os.clock()
	while Live.Running do
		local mr, root = rootOf(m), getRoot()
		if not (mr and root and alive(m) and alive(LocalPlayer) and toolNamed("Gun")) then return false end
		if strict and ((mr.Position - root.Position).Magnitude > M.shootRange + 8 or not lineOfSight(mr)) then return false end
		local waited = os.clock() - t0
		local pos, conf, state = solve(m, lagTime() + M.leadT)
		local warm = #trackerOf(m).s >= 6
		if warm and pos and (not M.waitGood or conf >= M.minConf or waited >= M.patience) then
			M.lastShot = os.clock()
			fireAt(gun, pos, conf, state)
			return true
		end
		if waited > M.patience + 1.5 then return false end
		task.wait(0.02)
	end
	return false
end
local function shootMurderer()
	if not toolNamed("Gun") then notify("You need the Gun (be the Sheriff)", T.Warn) return end
	local m = findRole("Murderer")
	if not (m and rootOf(m)) then notify("No murderer to shoot", T.Warn) return end
	task.spawn(function()
		if not takeShot(m, false) then notify("Couldn't take the shot", T.Warn) end
	end)
end
-- rage: stick directly behind the murderer, keep the camera locked on them and keep firing until they die or the gun is lost
local function rageShoot()
	runSeq(function(ok)
		local r = getRoot()
		if not r then return end
		local gun0 = toolNamed("Gun")
		if gun0 then equipGun(gun0) end
		local home, last, shots = r.CFrame, 0, 0
		local start = os.clock()
		-- after teleporting behind them the server still has us at the old spot for a moment: do not fire until it has caught up
		local settle = math.clamp(pingTime() * 0.6 + 0.08, 0.1, 0.5)
		local waitSince = start + settle
		while ok() do
			local gun, root = toolNamed("Gun"), getRoot()
			local m = findRole("Murderer")
			local mr = rootOf(m)
			if not gun or not root or not mr or m == LocalPlayer or not alive(LocalPlayer) then break end
			trackerOf(m)
			local back = mr.CFrame * CFrame.new(0, 0, M.rageDist)
			root.CFrame = CFrame.lookAt(back.Position, mr.Position)
			root.AssemblyLinearVelocity = Vector3.zero
			if gun.Parent ~= LocalPlayer.Character then equip(gun) end
			local c = cam()
			if c then c.CFrame = CFrame.lookAt(c.CFrame.Position, mr.Position) end
			local now = os.clock()
			if gun.Parent == LocalPlayer.Character and now - start >= settle and now - last >= M.rageGap then
				local pos, conf, state = solve(m, lagTime())
				if pos and #trackerOf(m).s >= 6 and (not M.waitGood or conf >= M.minConf or now - waitSince >= M.patience) then
					last, waitSince, shots = now, now, shots + 1
					fireAt(gun, pos, conf, state)
				end
			end
			RunService.Heartbeat:Wait()
		end
		local root = getRoot()
		if root and alive(LocalPlayer) then root.CFrame = home root.AssemblyLinearVelocity = Vector3.zero end
		if shots > 0 then notify("Rage shoot finished (" .. shots .. " shots)", T.Sub) end
	end)
end
task.spawn(function()
	while Live.Running do
		if M.autoRage and not M.busy and toolNamed("Gun") and alive(LocalPlayer) and findRole("Murderer") then rageShoot() task.wait(0.5)
		elseif M.autoShoot and not M.autoRage and toolNamed("Gun") and alive(LocalPlayer) then
			-- only take shots that are likely to land: close enough, a clear line of sight and (see takeShot) a predictable target
			local m, r = findRole("Murderer"), getRoot()
			local mr = rootOf(m)
			if mr and r and m ~= LocalPlayer and (mr.Position - r.Position).Magnitude <= M.shootRange and lineOfSight(mr) then
				local gun = toolNamed("Gun")
				if M.keepGun and gun and gun.Parent ~= LocalPlayer.Character then equip(gun) end -- already in hand when the moment comes
				if os.clock() - M.lastShot >= M.shotCd and takeShot(m, true) and M.retreat then
					task.wait(0.4)
					local m2 = findRole("Murderer")
					local mr2, r2 = rootOf(m2), getRoot()
					if mr2 and r2 and alive(m2) and (mr2.Position - r2.Position).Magnitude < 45 then awayFromMurderer(70) end
				end
			end
		end
		task.wait(0.3)
	end
end)

	local i, target = tpDd.Get(), nil
	if i == 1 then
		target = findRole("Murderer")
	elseif i == 2 then
		target = findRole("Sheriff")

	else
		local pool = {}
		for _, p in ipairs(Players:GetPlayers()) do if p ~= LocalPlayer and alive(p) and rootOf(p) then table.insert(pool, p) end end
		target = pool[math.random(math.max(#pool, 1))]
	end
	local tr, r = rootOf(target), getRoot()
	if tr and r then r.CFrame = tr.CFrame * CFrame.new(0, 0, 4) else notify("No " .. tpDd.Name():lower() .. " found", T.Warn) end
