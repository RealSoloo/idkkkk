-- ---------------------------------------------------- tools / hotbar slots
local slots = {}
local slotRow
local function snapshot()
	table.clear(slots)
	local bp = LocalPlayer:FindFirstChildOfClass("Backpack")
	if bp then for _, c in ipairs(bp:GetChildren()) do if c:IsA("Tool") then table.insert(slots, c.Name) end end end
	local parts = {}
	for i, n in ipairs(slots) do table.insert(parts, i .. ":" .. n) end
	if slotRow then slotRow.Text = "Slots  " .. (#parts > 0 and table.concat(parts, "   ") or "(unequip your tool, then press Refresh)") end
end
local function toolNamed(n)
	local ch, bp = getChar(), LocalPlayer:FindFirstChildOfClass("Backpack")
	local t = (ch and ch:FindFirstChild(n)) or (bp and bp:FindFirstChild(n))
	return t and t:IsA("Tool") and t or nil
end
local function slotTool(n) local nm = slots[n] return nm and toolNamed(nm) or nil end
local function allTools()
	local list, ch, bp = {}, getChar(), LocalPlayer:FindFirstChildOfClass("Backpack")
	for _, src in ipairs({ ch, bp }) do
		for _, c in ipairs(src and src:GetChildren() or {}) do if c:IsA("Tool") then table.insert(list, c) end end
	end
	return list
end
local function equip(tool)
	if tool.Parent ~= getChar() then
		local h = getHum()
		if h then h:EquipTool(tool) else tool.Parent = getChar() end
	end
end
local function getBomb() return toolNamed("Timebomb") or slotTool(DS.bombSlot) end
local function getWeapon()
	local m = DS.weapMode
	if m == 1 then
		local t = slotTool(DS.weapSlot)
		if t then return t end
		for _, c in ipairs(allTools()) do if c.Name:lower():find("sword", 1, true) then return c end end
		return nil
	elseif m == 2 then
		local ch = getChar()
		return ch and ch:FindFirstChildOfClass("Tool")
	elseif m == 3 then
		for _, c in ipairs(allTools()) do if c.Name:lower():find(DS.weapName:lower(), 1, true) then return c end end
		return nil
	else
		local list = allTools()
		if #list == 0 then return nil end
		DS.rot = DS.rot % #list + 1
		return list[DS.rot]
	end
end
local function swing()
	local tool = getWeapon()
	if not tool then return false end
	equip(tool)
	pcall(function() tool:Activate() end)
	return true
end

-- ---------------------------------------------------- helpers
local lastWarn = 0
local function warn2(text)
	if os.clock() - lastWarn > 2 then lastWarn = os.clock() notify(text, T.Warn) end
end
local function alive(p)
	local h = p and p.Character and p.Character:FindFirstChildOfClass("Humanoid")
	return h ~= nil and h.Health > 0
end
local function rootOf(p) return p and p.Character and p.Character:FindFirstChild("HumanoidRootPart") end
local function hpOf(p)
	local h = p and p.Character and p.Character:FindFirstChildOfClass("Humanoid")
	return h and h.Health or 0
end
local function stopAll() DS.token = DS.token + 1 DS.busy = false end
local function runSeq(fn)
	if DS.busy then notify("Already running - press Stop first", T.Warn) return end
	DS.token = DS.token + 1
	local my = DS.token
	DS.busy = true
	task.spawn(function()
		local ok = function() return Live.Running and DS.token == my end
		local good, err = pcall(fn, ok)
		if not good then warn2("Error: " .. tostring(err):sub(1, 60)) end
		if DS.token == my then DS.busy = false end
	end)
end
local function enemies()
	local list = {}
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LocalPlayer and alive(p) and rootOf(p) and not (DS.skipTeam and S.SameTeam(p)) then table.insert(list, p) end
	end
	local r = getRoot()
	if DS.prio == 2 then
		table.sort(list, function(a, b) return hpOf(a) < hpOf(b) end)
	elseif DS.prio == 3 then
		table.sort(list, function(a, b) return hpOf(a) > hpOf(b) end)
	elseif r then
		table.sort(list, function(a, b) return (rootOf(a).Position - r.Position).Magnitude < (rootOf(b).Position - r.Position).Magnitude end)
	end
	return list
end
local function goHome(home)
	local r = getRoot()
	if r and DS.home and home then r.CFrame = home r.AssemblyLinearVelocity = Vector3.zero end
end

-- ---------------------------------------------------- bombs
local function prepBomb()
	local tool = getBomb()
	local ev = tool and tool:FindFirstChild("Fire")
	if not ev then warn2("Timebomb not found (check the slot numbers)") return nil end
	if DS.equip then equip(tool) end
	return ev
end
local function fireAt(ev, pos)
	pcall(function() ev:FireServer(CFrame.new(pos + Vector3.new(0, DS.height, 0))) end)
end
local function bombSpam(ev, pos, count, ok)
	for _ = 1, count do
		if not ok() then return end
		fireAt(ev, pos)
		task.wait(math.max(DS.gap, 0.01))
	end
end
local function targetPos(mode)
	if mode == 1 then
		local hit = LocalPlayer:GetMouse().Hit
		return hit and hit.Position
	elseif mode == 2 then
		local r = getRoot()
		return r and r.Position
	elseif mode == 3 then
		local tr = rootOf(S.Target)
		if not tr then return nil end
		if DS.lead then return tr.Position + tr.AssemblyLinearVelocity * DS.leadT end
		return tr.Position
	else
		local p = nearestPlayer()
		local tr = rootOf(p)
		if not tr then return nil end
		if DS.lead then return tr.Position + tr.AssemblyLinearVelocity * DS.leadT end
		return tr.Position
	end
end
local function drop(mode)
	local pos = targetPos(mode or DS.mode)
	if not pos then warn2("No drop position (" .. MODES[mode or DS.mode] .. ")") return end
	local ev = prepBomb()
	if ev then fireAt(ev, pos) end
end
S.DoomDrop = drop

-- enemy team spawns
local function enemySpawns()
	local out, anyTeam = {}, false
	local color = LocalPlayer.TeamColor
	local hasTeam = LocalPlayer.Team ~= nil
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("SpawnLocation") and not d.Neutral then
			anyTeam = true
			if not hasTeam or d.TeamColor ~= color then table.insert(out, d) end
		end
	end
	if not anyTeam then
		-- fallback: parts called "...spawn..." that aren't the colour of your team
		for _, d in ipairs(Workspace:GetDescendants()) do
			if d:IsA("BasePart") and d.Name:lower():find("spawn", 1, true) and not d:IsDescendantOf(getChar() or Workspace) and d.BrickColor ~= color then
				table.insert(out, d)
			end
		end
	end
	return out
end

-- ---------------------------------------------------- weapon attack
local function attack(p, ok, maxTime)
	local t0, gap = os.clock(), 1 / math.max(DS.cps, 1)
	while ok() and alive(p) and os.clock() - t0 < (maxTime or DS.timeout) do
		local r, tr = getRoot(), rootOf(p)
		if not r or not tr then break end
		local back = tr.CFrame * CFrame.new(0, 0, DS.dist)
		r.CFrame = CFrame.lookAt(back.Position, tr.Position)
		r.AssemblyLinearVelocity = Vector3.zero
		if not swing() then warn2("No weapon found (check weapon slot / mode)") return false end
		task.wait(gap)
	end
	return not alive(p)
end
local function killList(list)
	runSeq(function(ok)
		local r = getRoot()
		if not r then return end
		local home, n = r.CFrame, 0
		for _, p in ipairs(list) do
			if not ok() then break end
			if alive(p) and attack(p, ok) then n = n + 1 end
		end
		goHome(home)
		notify(("Done - %d of %d eliminated"):format(n, #list), n > 0 and T.Good or T.Warn)
	end)
end
local function killAll() killList(enemies()) end
local function killSelected()
	if S.Target and S.Target.Parent then killList({ S.Target }) else notify("Select a player in the Players tab first", T.Warn) end
end
local function killNearest()
	local p = enemies()[1]
	if p then killList({ p }) else notify("No enemies found", T.Warn) end
end

-- ---------------------------------------------------- loops
task.spawn(function() -- auto bomb loop
	while Live.Running do
		if DS.loop then drop() task.wait(math.max(DS.delay, 0.03)) else task.wait(0.1) end
	end
end)
task.spawn(function() -- kill aura / auto click / loop kills
	while Live.Running do
		local gap = 1 / math.max(DS.cps, 1)
		if DS.aura and not DS.busy then
			local r = getRoot()
			if r then
				for _, p in ipairs(enemies()) do
					if (rootOf(p).Position - r.Position).Magnitude <= DS.auraR then swing() break end
				end
			end
		end
		if DS.autoClick and not DS.busy then
			local ch = getChar()
			local t = ch and ch:FindFirstChildOfClass("Tool")
			if t then pcall(function() t:Activate() end) end
		end
		if DS.loopAll and not DS.busy then killAll() end
		if DS.loopSel and not DS.busy and S.Target and alive(S.Target) then killList({ S.Target }) end
		task.wait(math.max(gap, 0.03))
	end
end)
-- sword reach (client side: bigger handle = bigger hit area)
local reachOrig = setmetatable({}, { __mode = "k" })
local function restoreReach()
	for h, o in pairs(reachOrig) do if h.Parent then h.Size, h.Massless = o.size, o.massless end end
	table.clear(reachOrig)
end
bind(RunService.Heartbeat, function()
	if not DS.reach then return end
	local ch = getChar()
	local tool = ch and ch:FindFirstChildOfClass("Tool")
	local h = tool and tool:FindFirstChild("Handle")
	if h and h:IsA("BasePart") then
		if not reachOrig[h] then reachOrig[h] = { size = h.Size, massless = h.Massless } end
		h.Size = Vector3.new(DS.reachSize, DS.reachSize, DS.reachSize)
		h.Massless = true
	end
end)

local liveBombs, bombHl, lastDodge = {}, {}, 0
local function isLiveBomb(d)
	if not (d:IsA("BasePart") or d:IsA("Model")) then return false end
	if not d.Name:lower():find("bomb", 1, true) then return false end
	local me, bp = getChar(), LocalPlayer:FindFirstChildOfClass("Backpack")
	if (me and d:IsDescendantOf(me)) or (bp and d:IsDescendantOf(bp)) then return false end
	return d:IsDescendantOf(Workspace)
end
for _, d in ipairs(Workspace:GetDescendants()) do if isLiveBomb(d) then liveBombs[d] = true end end
bind(Workspace.DescendantAdded, function(d)
	if d.Name:lower():find("bomb", 1, true) then
		task.defer(function() if isLiveBomb(d) then liveBombs[d] = true end end)
	end
end)
local function bombPos(b)
	if b:IsA("BasePart") then return b.Position end
	local ok, pv = pcall(function() return b:GetPivot().Position end)
	return ok and pv or nil
end
task.spawn(function() -- bomb ESP + auto dodge
	while Live.Running do
		for b in pairs(liveBombs) do
			if not b.Parent then
				liveBombs[b] = nil
				if bombHl[b] then bombHl[b]:Destroy() bombHl[b] = nil end
			elseif DS.bombEsp and not bombHl[b] then
				bombHl[b] = new("Highlight", { Adornee = b, FillColor = Color3.fromRGB(255, 70, 70), OutlineColor = Color3.fromRGB(255, 255, 255), FillTransparency = 0.3, DepthMode = Enum.HighlightDepthMode.AlwaysOnTop }, EspGui)
			end
		end
		if not DS.bombEsp then
			for b, h in pairs(bombHl) do h:Destroy() bombHl[b] = nil end
		end
		local r = getRoot()
		if DS.dodge and r and os.clock() - lastDodge > 0.6 then
			for b in pairs(liveBombs) do
				local bp = bombPos(b)
				if bp and (bp - r.Position).Magnitude <= DS.dodgeR then
					local away = r.Position - bp
					away = Vector3.new(away.X, 0, away.Z)
					away = away.Magnitude > 0.5 and away.Unit or Vector3.new(1, 0, 0)
					r.CFrame = r.CFrame + away * (DS.dodgeR + 8)
					r.AssemblyLinearVelocity = Vector3.zero
					lastDodge = os.clock()
					break
				end
			end
		end
		task.wait(0.1)
	end
end)

-- bomb patterns
local function randomIn(radius)
	local a, d = math.random() * math.pi * 2, math.sqrt(math.random()) * radius
	return Vector3.new(math.cos(a) * d, 0, math.sin(a) * d)
end
local function bombEveryone()
	runSeq(function(ok)
		local ev = prepBomb()
		if not ev then return end
		for _, p in ipairs(enemies()) do
			if not ok() then break end
			local tr = rootOf(p)
			if tr then bombSpam(ev, tr.Position + tr.AssemblyLinearVelocity * (DS.lead and DS.leadT or 0), DS.bombCount, ok) end
		end
	end)
end
local function raidSpawns()
	runSeq(function(ok)
		local r = getRoot()
		if not r then return end
		local spawns = enemySpawns()
		if #spawns == 0 then notify("No enemy spawns found", T.Warn) return end
		local ev = prepBomb()
		if not ev then return end
		local home = r.CFrame
		for _, sp in ipairs(spawns) do
			if not ok() then break end
			if not DS.noTp then
				local root = getRoot()
				if not root then break end
				root.CFrame = sp.CFrame + Vector3.new(0, 6, 0)
				root.AssemblyLinearVelocity = Vector3.zero
				task.wait(0.15)
			end
			bombSpam(ev, sp.Position, DS.bombCount, ok)
		end
		if not DS.noTp then goHome(home) end
	end)
end
task.spawn(function()
	while Live.Running do
		if DS.orbit and not DS.busy then
			local r, ev = getRoot(), getBomb() and getBomb():FindFirstChild("Fire")
			if r and ev then
				DS.orbitA = DS.orbitA + 0.7
				if DS.equip then equip(getBomb()) end
				fireAt(ev, r.Position + Vector3.new(math.cos(DS.orbitA) * DS.ringR, 0, math.sin(DS.orbitA) * DS.ringR))
			end
		end
		if DS.loopBombAll and not DS.busy then bombEveryone() end
		if DS.spawnCamp and not DS.busy then raidSpawns() end
		if DS.stick and not DS.busy and S.Target and alive(S.Target) then
			local r, tr = getRoot(), rootOf(S.Target)
			if r and tr then
				local back = tr.CFrame * CFrame.new(0, 0, DS.dist)
				r.CFrame = CFrame.lookAt(back.Position, tr.Position)
				r.AssemblyLinearVelocity = Vector3.zero
				swing()
			end
		end
		local w = 0.1
		if DS.orbit then w = DS.delay end
		task.wait(math.max(w, 0.05))
	end
end)
onUnload(function()
	DS.orbit, DS.loopBombAll, DS.spawnCamp, DS.stick, DS.dodge, DS.bombEsp = false, false, false, false, false, false
	for b, h in pairs(bombHl) do h:Destroy() end
	table.clear(bombHl)
end)
