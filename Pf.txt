-- ---------------------------------------------------- helpers
local function playersRoot() return Workspace:FindFirstChild("Players") end
local function findPart(m, names)
	for _, n in ipairs(names) do
		local p = m:FindFirstChild(n)
		if p and p:IsA("BasePart") then return p end
	end
	-- Part casing differs between character versions. Do not recursively match
	-- weapons/accessories or turn a team container into a character.
	for _, p in ipairs(m:GetChildren()) do
		if p:IsA("BasePart") then
			for _, n in ipairs(names) do
				if p.Name:lower() == n:lower() then return p end
			end
		end
	end
	return nil
end
local function firstPart(m)
	if m.PrimaryPart then return m.PrimaryPart end
	for _, ch in ipairs(m:GetChildren()) do
		if ch:IsA("BasePart") then return ch end
	end
	return nil
end
local HEAD_NAMES, TORSO_NAMES = { "Head", "head" }, { "Torso", "UpperTorso", "HumanoidRootPart", "torso", "Chest" }
local function headOf(m)
	local named = findPart(m, HEAD_NAMES)
	if named then return named end
	-- Randomized PF models still keep their direct body parts. The highest
	-- body-part center is a usable head fallback without matching weapon parts.
	local best
	for _, part in ipairs(m:GetChildren()) do
		if part:IsA("BasePart") and (not best or part.Position.Y > best.Position.Y) then best = part end
	end
	return best or firstPart(m)
end
local function torsoOf(m)
	local named = findPart(m, TORSO_NAMES)
	if named then return named end
	local best, volume = nil, -1
	for _, part in ipairs(m:GetChildren()) do
		if part:IsA("BasePart") then
			local size = part.Size
			local v = size.X * size.Y * size.Z
			if v > volume then best, volume = part, v end
		end
	end
	return best or firstPart(m)
end
local function teamKey(value)
	return tostring(value or ""):lower():gsub("[%s_%-]", "")
end
-- PF uses faction names in some places and BrickColor folder names in others.
-- Compare normalized aliases, not just the literal Player.Team.Name.
local TEAM_ALIASES = {
	phantoms = { "Phantoms", "Phantom", "Bright blue" },
	ghosts = { "Ghosts", "Ghost", "Bright orange" },
}
local function addTeamName(set, value)
	if not value then return end
	local key = teamKey(value)
	if key == "" then return end
	set[key] = true
	for _, names in pairs(TEAM_ALIASES) do
		local matches = false
		for _, name in ipairs(names) do
			if key == teamKey(name) then matches = true break end
		end
		if matches then
			for _, name in ipairs(names) do set[teamKey(name)] = true end
		end
	end
end
local function factionOf(value)
	local key = teamKey(value)
	for faction, names in pairs(TEAM_ALIASES) do
		for _, name in ipairs(names) do
			if key == teamKey(name) then return faction end
		end
	end
	return nil
end
-- Read the game's own leaderboard; never use this hub's labels as evidence.
local roster, rosterAt, rosterChecked = {}, -math.huge, -math.huge
local owners = setmetatable({}, { __mode = "k" })
local replication, replicationChecked = nil, -math.huge
local function plainText(value)
	return tostring(value or ""):gsub("<[^>]*>", ""):match("^%s*(.-)%s*$"):lower()
end
local function gameGui(node)
	local cur = node
	while cur and cur ~= PlayerGui do
		if cur.Name:sub(1, 5) == "Solos" then return false end
		if cur:IsA("GuiObject") and not cur.Visible then return false end
		if cur:IsA("ScreenGui") and not cur.Enabled then return false end
		cur = cur.Parent
	end
	return cur == PlayerGui
end
local function commonParent(a, b)
	local cur = a.Parent
	while cur and cur ~= PlayerGui do
		if b:IsDescendantOf(cur) then return cur end
		cur = cur.Parent
	end
	return nil
end
local function headerBranch(header, root)
	local cur = header
	while cur.Parent and cur.Parent ~= root do cur = cur.Parent end
	return cur
end
local function readScoreboard()
	if not PlayerGui then return end
	local now = os.clock()
	if now - rosterChecked < 0.75 then return end
	rosterChecked = now
	local phantoms, ghosts, labels = {}, {}, {}
	local playersByText, ambiguous = {}, {}
	for _, pl in ipairs(Players:GetPlayers()) do
		playersByText[plainText(pl.Name)] = pl
	end
	for _, pl in ipairs(Players:GetPlayers()) do
		local key = plainText(pl.DisplayName)
		if playersByText[key] and playersByText[key] ~= pl then ambiguous[key] = true
		elseif not ambiguous[key] then playersByText[key] = pl end
	end
	for key in pairs(ambiguous) do playersByText[key] = nil end
	for _, node in ipairs(PlayerGui:GetDescendants()) do
		if (node:IsA("TextLabel") or node:IsA("TextButton")) and gameGui(node) then
			local text = plainText(node.Text)
			if text == "phantoms" then table.insert(phantoms, node)
			elseif text == "ghosts" then table.insert(ghosts, node)
			elseif playersByText[text] then table.insert(labels, { node = node, player = playersByText[text] }) end
		end
	end
	for _, ph in ipairs(phantoms) do
		for _, gh in ipairs(ghosts) do
			local root = commonParent(ph, gh)
			if root then
				local left, right = headerBranch(ph, root), headerBranch(gh, root)
				local found, conflicts = {}, {}
				for _, row in ipairs(labels) do
					local node, faction = row.node, nil
					if node:IsDescendantOf(root) then
						if left ~= right and node:IsDescendantOf(left) then faction = "phantoms"
						elseif left ~= right and node:IsDescendantOf(right) then faction = "ghosts"
						else
							-- Some scoreboards keep the header and row containers separate.
							local a, b, n = ph.AbsolutePosition, gh.AbsolutePosition, node.AbsolutePosition
							if math.abs(a.Y - b.Y) <= 48 and math.abs(a.X - b.X) > 100 and
								n.Y > math.max(a.Y, b.Y) and n.X >= math.min(a.X, b.X) then
								local split = (a.X + b.X) / 2
								faction = (n.X < split) == (a.X < b.X) and "phantoms" or "ghosts"
							end
						end
					end
					if faction then
						if found[row.player] and found[row.player] ~= faction then conflicts[row.player] = true end
						found[row.player] = faction
					end
				end
				for pl in pairs(conflicts) do found[pl] = nil end
				if next(found) then
					roster, rosterAt = found, now
					local n = 0 for _ in pairs(found) do n = n + 1 end
					P.rosterStatus = "leaderboard: " .. n .. " players"
					return
				end
			end
		end
	end
	P.rosterStatus = now - rosterAt <= 20 and "leaderboard cached (20s maximum)" or "open the game's leaderboard to read teams"
end
local function rosterFaction(pl)
	if os.clock() - rosterAt <= 20 then return roster[pl] end
	return nil
end
local function getReplication()
	if replication then return replication end
	if os.clock() - replicationChecked < 5 then return nil end
	replicationChecked = os.clock()
	local sources = {}
	local function add(value) if type(value) == "table" then table.insert(sources, value) end end
	add(shared)
	add(env.shared)
	local getRuntime = env.getrenv or getrenv
	if type(getRuntime) == "function" then
		local ok, runtime = pcall(getRuntime)
		if ok and type(runtime) == "table" then add(runtime.shared) end
	end
	for _, source in ipairs(sources) do
		local loader = rawget(source, "require")
		if type(loader) == "function" then
			local ok, module = pcall(loader, "ReplicationInterface")
			if ok and type(module) == "table" and type(module.getEntry) == "function" then
				replication = module
				return module
			end
		end
	end
	P.ownerStatus = "replication interface unavailable; randomized bodies cannot be linked to usernames"
	return nil
end
local function replicationModel(entry)
	if type(entry) ~= "table" then return nil end
	local third = entry._thirdPersonObject
	if type(entry.getThirdPersonObject) == "function" then third = entry:getThirdPersonObject() end
	if type(third) ~= "table" then return nil end
	local model = third._character or third._characterModel
	if type(third.getCharacterModel) == "function" then model = third:getCharacterModel() end
	if typeof(model) == "Instance" and model:IsA("Model") then return model end
	return nil
end
local function refreshOwnership()
	local fresh = setmetatable({}, { __mode = "k" })
	local module = getReplication()
	local count = 0
	if module then
		for _, pl in ipairs(Players:GetPlayers()) do
			local ok, model = pcall(function() return replicationModel(module.getEntry(pl)) end)
			if ok and model and model.Parent then fresh[model] = pl count = count + 1 end
		end
		P.ownerStatus = "replication interface: " .. count .. " model owners"
	end
	owners = fresh
end
local function playerFaction(pl)
	local fromBoard = rosterFaction(pl)
	if fromBoard then return fromBoard end
	if pl.Team then
		local faction = factionOf(pl.Team.Name)
		if faction then return faction end
		if pl.Team.TeamColor then
			faction = factionOf(pl.Team.TeamColor.Name)
			if faction then return faction end
		end
	end
	-- PF can leave Neutral true and Team nil while still publishing its faction color.
	if pl.TeamColor then return factionOf(pl.TeamColor.Name) end
	return nil
end
local function folderFaction(m, root)
	local cur = m.Parent
	while cur and cur ~= root and cur ~= Workspace do
		local faction = factionOf(cur.Name)
		if faction then return faction, cur.Name end
		cur = cur.Parent
	end
	return nil
end
local function myFaction()
	if P.teamMode == 2 then return "phantoms", "manual" end
	if P.teamMode == 3 then return "ghosts", "manual" end
	local faction = playerFaction(LocalPlayer)
	if faction then return faction, "player team" end
	local character, root = LocalPlayer.Character, playersRoot()
	if character and root and character:IsDescendantOf(root) then
		faction = folderFaction(character, root)
		if faction then return faction, "local character folder" end
	end
	return nil, "unresolved - select Your team on the ESP tab"
end
local function myTeamNames()
	local set = {}
	local faction = myFaction()
	if faction then
		addTeamName(set, faction)
	else
		if LocalPlayer.Team then addTeamName(set, LocalPlayer.Team.Name) end
		if not LocalPlayer.Neutral and LocalPlayer.TeamColor then
			addTeamName(set, LocalPlayer.TeamColor.Name)
		end
	end
	return set
end
local function sameKnownTeam(pl)
	local mine, theirs = myFaction(), playerFaction(pl)
	if not mine or not theirs then return nil end
	return mine == theirs
end
local function modelPlayer(m)
	local pl = owners[m] or Players:GetPlayerFromCharacter(m) or Players:FindFirstChild(m.Name)
	return pl and pl:IsA("Player") and pl or nil
end
local function looksLikeChar(m, loose)
	if not m:IsA("Model") or m == LocalPlayer.Character or m:IsDescendantOf(cam()) then return false end
	if loose then return firstPart(m) ~= nil end
	return findPart(m, HEAD_NAMES) ~= nil and findPart(m, TORSO_NAMES) ~= nil
end
-- scan result is cached (refreshed by the loop below) so the per-frame code stays cheap.
-- Team is taken from the Workspace.Players team folder (own team folder = teammates); falls back to the player's team.
local cache, cacheSet = {}, {}
local function groupOf(m)
	local root, cur = playersRoot(), m
	if not root then return nil end
	while cur.Parent and cur.Parent ~= root do cur = cur.Parent end
	if cur.Parent == root and cur ~= m and cur:IsA("Folder") then return cur end
	return nil
end
local function calibratedEnemy(group)
	local root = playersRoot()
	if P.myGroup and P.myGroup.Parent ~= root then P.myGroup = nil end
	if P.enemyGroup and P.enemyGroup.Parent ~= root then P.enemyGroup = nil end
	if not group then return nil end
	if group == P.myGroup then return false end
	if group == P.enemyGroup then return true end
	-- A complementary team is certain only when there are exactly two groups.
	local groups = {}
	if root then
		for _, child in ipairs(root:GetChildren()) do
			if child:IsA("Folder") then table.insert(groups, child) end
		end
	end
	if #groups == 2 and (group == groups[1] or group == groups[2]) then
		if P.myGroup then return true end
		if P.enemyGroup then return false end
	end
	return nil
end
local function scan(loose)
	P.scanErr = nil
	local ok, err = pcall(readScoreboard)
	if not ok then P.rosterStatus = "leaderboard read failed: " .. tostring(err) end
	ok, err = pcall(refreshOwnership)
	if not ok then P.ownerStatus = "owner lookup failed: " .. tostring(err) end
	local root, mine = playersRoot(), myTeamNames()
	local faction, source = myFaction()
	P.teamFaction, P.teamSource = faction, source
	if loose and not root then return end
	local bases = {}
	if root then table.insert(bases, root) end
	local roots = Workspace:FindFirstChild("Roots")
	if roots and not loose then table.insert(bases, roots) end
	if #bases == 0 then table.insert(bases, Workspace) end
	local found, set, hasMine = {}, {}, false
	if root then
		for _, f in ipairs(root:GetChildren()) do
			if mine[teamKey(f.Name)] then hasMine = true end
		end
	end
	for _, base in ipairs(bases) do
	for _, d in ipairs(base:GetDescendants()) do
		local ok, err = pcall(function()
			if set[d] or d.Name == LocalPlayer.Name or not looksLikeChar(d, loose) then return end
			if d.Parent and d.Parent:IsA("Model") and looksLikeChar(d.Parent, loose) then return end
			local top = d
			while top.Parent and top.Parent ~= base do top = top.Parent end
			local folderName = (top ~= d) and top.Name or (d.Parent and d.Parent.Name) or ""
			local enemy = nil
			local pl = modelPlayer(d)
			if pl == LocalPlayer then return end
			local modelFaction, factionFolder = folderFaction(d, root)
			if factionFolder then folderName = factionFolder end
			-- A manual faction selection works even when Roblox Player.Team is
			-- unset, and does not need our faction's folder to contain a player.
			local same = nil
			if pl then same = sameKnownTeam(pl) end
			local group = groupOf(d)
			local calibrated = calibratedEnemy(group)
			if calibrated ~= nil then
				enemy = calibrated
			elseif same ~= nil then
				enemy = not same
			elseif faction and modelFaction then
				enemy = faction ~= modelFaction
			elseif hasMine then
				enemy = not mine[teamKey(folderName)]
			end
			local e = { m = d, enemy = enemy, player = pl, folder = folderName, group = group }
			table.insert(found, e)
			set[d] = e
		end)
		if not ok then P.scanErr = tostring(err) end
	end
	end
	local groupFactions, groupConflict = {}, {}
	for _, entry in ipairs(found) do
		local f = entry.player and playerFaction(entry.player)
		if entry.group and f then
			if groupFactions[entry.group] and groupFactions[entry.group] ~= f then groupConflict[entry.group] = true end
			groupFactions[entry.group] = f
		end
	end
	for _, entry in ipairs(found) do
		local f = entry.group and not groupConflict[entry.group] and groupFactions[entry.group]
		if entry.enemy == nil and faction and f then entry.enemy = faction ~= f end
	end
	cache, cacheSet = found, set
end
local function collect(includeTeam, includeUnknown)
	local out = {}
	for _, e in ipairs(cache) do
		if e.m.Parent and (e.enemy or includeTeam or P.allEnemy or (includeUnknown and e.enemy == nil)) then table.insert(out, e) end
	end
	return out
end
local function collectEsp()
	local out = {}
	for _, e in ipairs(cache) do
		if e.m.Parent and (not P.teamCheck or e.enemy == true) then
			table.insert(out, e)
		end
	end
	return out
end
-- Calibration must not depend on CanQuery: replicated PF body parts can be
-- visible but absent from Workspace:Raycast results. Intersect the center ray
-- with their oriented boxes directly, then allow a small screen-space margin.
local function bodyRayDistance(part, origin, direction, limit)
	local o = part.CFrame:PointToObjectSpace(origin)
	local d = part.CFrame:VectorToObjectSpace(direction)
	local near, far = 0, limit
	for _, axis in ipairs({ "X", "Y", "Z" }) do
		local half = part.Size[axis] / 2
		if math.abs(d[axis]) < 1e-8 then
			if o[axis] < -half or o[axis] > half then return nil end
		else
			local a, b = (-half - o[axis]) / d[axis], (half - o[axis]) / d[axis]
			if a > b then a, b = b, a end
			near, far = math.max(near, a), math.min(far, b)
			if near > far then return nil end
		end
	end
	return near
end
local function calibrationCandidate(c)
	if not c.ViewportSize then return nil end
	local center = Vector2.new(c.ViewportSize.X / 2, c.ViewportSize.Y / 2)
	local ray = c:ViewportPointToRay(center.X, center.Y)
	local best, bestDistance, close, closePixels, closeDepth = nil, P.espMax, nil, 16, math.huge
	for _, entry in ipairs(cache) do
		if entry.group and entry.m.Parent then
			-- Ignore nested weapon/accessory geometry. Only the direct body parts
			-- count; their names and query/collision flags do not matter.
			for _, part in ipairs(entry.m:GetChildren()) do
				if part:IsA("BasePart") then
					local ok, distance = pcall(bodyRayDistance, part, ray.Origin, ray.Direction, P.espMax)
					if ok and distance and distance < bestDistance then best, bestDistance = entry, distance end
					local sp, on = c:WorldToViewportPoint(part.Position)
					if on and sp.Z > 0 and sp.Z <= P.espMax then
						local pixels = (Vector2.new(sp.X, sp.Y) - center).Magnitude
						if pixels < closePixels or (pixels == closePixels and sp.Z < closeDepth) then
							close, closePixels, closeDepth = entry, pixels, sp.Z
						end
					end
				end
			end
		end
	end
	return best or close
end
local function calibrateTeam(isEnemy)
	local c = cam()
	if not c then return notify("Camera is not ready", T.Warn) end
	scan(false)
	if #cache == 0 then scan(true) end
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	local ignore = { c }
	local ignored = Workspace:FindFirstChild("Ignore")
	if ignored then table.insert(ignore, ignored) end
	if LocalPlayer.Character then table.insert(ignore, LocalPlayer.Character) end
	params.FilterDescendantsInstances = ignore
	local hit = Workspace:Raycast(c.CFrame.Position, c.CFrame.LookVector * P.espMax, params)
	local cur = hit and hit.Instance
	local entry
	while cur and cur ~= Workspace do
		if cacheSet[cur] then entry = cacheSet[cur] break end
		cur = cur.Parent
	end
	if not entry or not entry.group then entry = calibrationCandidate(c) end
	if not entry or not entry.group then
		return notify(#cache == 0 and "No character models found yet - deploy and try again" or
			"Put a player's body at the center of the screen, then press F6 or F7", T.Warn)
	end
	P.myGroup, P.enemyGroup = nil, nil
	if isEnemy then P.enemyGroup = entry.group else P.myGroup = entry.group end
	P.target = nil
	scan(false)
	if #cache == 0 then scan(true) end
	notify(isEnemy and "Enemy group marked - team check updated" or "Teammate group marked - team check updated", T.Good)
end
local function enemyModelOf(inst)
	local cur = inst
	while cur and cur ~= Workspace do
		local e = cacheSet[cur]
		if e then return (e.enemy or P.allEnemy) and cur or nil end
		cur = cur.Parent
	end
	return nil
end
local rp = RaycastParams.new()
rp.FilterType = Enum.RaycastFilterType.Exclude
local function refreshFilter()
	local list = { cam() }
	local ig = Workspace:FindFirstChild("Ignore")
	if ig then table.insert(list, ig) end
	for _, e in ipairs(cache) do
		if e.enemy == false and e.m.Parent then table.insert(list, e.m) end
	end
	if LocalPlayer.Character then table.insert(list, LocalPlayer.Character) end
	rp.FilterDescendantsInstances = list
end
local function visibleModel(m, part)
	local c = cam()
	local origin = c.CFrame.Position
	local res = Workspace:Raycast(origin, part.Position - origin, rp)
	if res == nil or res.Instance:IsDescendantOf(m) then return true end
	local hit = res.Instance
	return hit.Transparency >= 0.9 or not hit.CanCollide
end
local function pickPart(m, center)
	if P.part == 1 then return headOf(m) or torsoOf(m) end
	if P.part == 2 then return torsoOf(m) or headOf(m) end
	local c, best, bestD = cam(), nil, math.huge
	for _, pt in ipairs({ headOf(m), torsoOf(m) }) do
		if pt then
			local sp = c:WorldToViewportPoint(pt.Position)
			local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
			if d < bestD then best, bestD = pt, d end
		end
	end
	return best
end
local function bestTarget(center)
	local c = cam()
	local best, bestPart, bestD = nil, nil, P.fov
	for _, e in ipairs(collect(false)) do
		local part = pickPart(e.m, center)
		if part then
			local sp, on = c:WorldToViewportPoint(part.Position)
			if on then
				local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
				local dist = (part.Position - c.CFrame.Position).Magnitude
				if d < bestD and dist <= P.maxDist and (not P.wall or visibleModel(e.m, part)) then best, bestPart, bestD = e.m, part, d end
			end
		end
	end
	return best, bestPart
end
local function aimActive()
	if UIS:GetFocusedTextBox() then return false end
	if P.mode == 3 then return true end
	if P.mode == 1 then return UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) end
	return P.key ~= Enum.KeyCode.Unknown and UIS:IsKeyDown(P.key)
end
local function press(down)
	if down and type(env.mouse1press) == "function" then pcall(env.mouse1press) return true end
	if not down and type(env.mouse1release) == "function" then pcall(env.mouse1release) return true end
	return false
end

local function updateAim(dt)
	P.aimErr = nil
	local c = cam()
	if not c then P.aimState = "waiting for camera" return end
	local vp = c.ViewportSize
	local center = Vector2.new(vp.X / 2, vp.Y / 2)
	fovGui.Visible = P.showFov and P.aim
	fovGui.Size = UDim2.fromOffset(P.fov * 2, P.fov * 2)
	fovGui.Position = UDim2.fromOffset(center.X, center.Y)
	fovStroke.Color = P.target and T.Good or T.Accent
	if not P.aim then P.target = nil P.aimState = "off" return end
	if not aimActive() then P.target = nil P.aimState = "waiting for ADS / key" return end
	refreshFilter()
	local m, part
	if P.sticky and P.target and P.target.Parent then
		local entry = cacheSet[P.target]
		if entry and (entry.enemy == true or P.allEnemy) then
			m = P.target
			part = pickPart(m, center)
		end
		local keep = false
		if part then
			local sp, on = c:WorldToViewportPoint(part.Position)
			local distance = (part.Position - c.CFrame.Position).Magnitude
			keep = on and distance <= P.maxDist and
				(Vector2.new(sp.X, sp.Y) - center).Magnitude <= P.fov * 1.3 and
				(not P.wall or visibleModel(m, part))
		end
		if not keep then P.target, m, part = nil, nil, nil end
	end
	if not m then
		m, part = bestTarget(center)
		if m then
			P.target, P.targetSince = m, os.clock()
			local j = P.jitter / 10
			P.offset = Vector3.new((math.random() - 0.5) * 2 * j, (math.random() - 0.5) * 2 * j, (math.random() - 0.5) * 2 * j)
		else
			P.target = nil
		end
	end
	if not (m and part) then
		local faction = myFaction()
		if #cache == 0 then
				P.aimState = "no character models found - copy diagnostic report"
			else
				local unresolved = false
				for _, entry in ipairs(cache) do
					if entry.enemy == nil then unresolved = true break end
				end
				P.aimState = unresolved and "mark teammate (F6) or enemy (F7) to calibrate team groups" or
					(not faction and "select Your team on ESP tab" or "no confirmed visible enemy in FOV / range")
			end
		return
	end
	if P.humanize and os.clock() - P.targetSince < P.delay then P.aimState = "reaction delay" return end
	local aimPos = part.Position
	if P.humanize then aimPos = aimPos + P.offset end
	local sp, on = c:WorldToViewportPoint(aimPos)
	if not on then P.aimState = "target off screen" return end
	local delta = Vector2.new(sp.X, sp.Y) - center
	if delta.Magnitude < P.deadzone then P.aimState = "on target" return end
	if P.aimMethod == 1 then
		-- Apply after the game's normal camera update so it cannot immediately
		-- overwrite the CFrame in the same frame. No mouse API is required.
		local alpha = 1 - (1 - math.clamp(P.strength, 0.01, 1)) ^ (math.min(dt or 1/60, 0.1) * 60)
		c.CFrame = c.CFrame:Lerp(CFrame.lookAt(c.CFrame.Position, aimPos), alpha)
		P.aimState = "tracking (camera)"
	else
		local move = env.mousemoverel or mousemoverel
		if type(move) ~= "function" then
			P.aimErr = "Mouse movement unavailable. Select Camera tracking."
			P.aimState = "mouse API unavailable"
			return
		end
		local step = delta * P.strength
		if step.Magnitude > P.maxStep then step = step.Unit * P.maxStep end
		local ok, err = pcall(move, step.X * P.scale, step.Y * P.scale)
		if not ok then
			P.aimErr = "Mouse movement failed: " .. tostring(err)
			P.aimState = "mouse API failed"
		else
			P.aimState = "tracking (mouse)"
		end
	end
end

	local ok, err = pcall(updateAim, dt)
	if not ok then P.aimErr = tostring(err) P.aimState = "error" end
end)

local ENEMY_VIS, ENEMY_HID = Color3.fromRGB(255, 70, 70), Color3.fromRGB(255, 190, 60)
-- Clip bounding-box edges to the near plane before projecting. Rejecting an
-- entire box because just one corner is behind the camera hides nearby players.
local function projectBounds(c, cf, size)
	local corners, points = {}, {}
	local near = 0.05
	for i = 0, 7 do
		local sx = (i % 2 == 0) and -1 or 1
		local sy = (math.floor(i / 2) % 2 == 0) and -1 or 1
		local sz = (i < 4) and -1 or 1
		local p = c.CFrame:PointToObjectSpace(cf:PointToWorldSpace(
			Vector3.new(sx * size.X / 2, sy * size.Y / 2, sz * size.Z / 2)))
		corners[i + 1] = p
		if p.Z <= -near then table.insert(points, p) end
	end
	for i = 0, 7 do
		for _, step in ipairs({1, 2, 4}) do
			if math.floor(i / step) % 2 == 0 then
				local a, b = corners[i + 1], corners[i + step + 1]
				if (a.Z <= -near) ~= (b.Z <= -near) then
					local t = (-near - a.Z) / (b.Z - a.Z)
					table.insert(points, a + (b - a) * t)
				end
			end
		end
	end
	local minX, minY, maxX, maxY = math.huge, math.huge, -math.huge, -math.huge
	for _, p in ipairs(points) do
		local sp = c:WorldToViewportPoint(c.CFrame:PointToWorldSpace(p))
		minX, minY = math.min(minX, sp.X), math.min(minY, sp.Y)
		maxX, maxY = math.max(maxX, sp.X), math.max(maxY, sp.Y)
	end
	local vp = c.ViewportSize
	if #points == 0 or maxX <= 0 or maxY <= 0 or minX >= vp.X or minY >= vp.Y then return nil end
	minX, minY = math.max(0, minX), math.max(0, minY)
	maxX, maxY = math.min(vp.X, maxX), math.min(vp.Y, maxY)
	if maxX <= minX or maxY <= minY then return nil end
	return minX, minY, maxX, maxY
end
local function drawModel(c, e)
	local m = e.m
	local cf, size = m:GetBoundingBox()
	local dist = (cf.Position - c.CFrame.Position).Magnitude
	local o = objs[m]
	if dist > P.espMax then
		if o then hideObj(o) end
		return false
	end
	local minX, minY, maxX, maxY = projectBounds(c, cf, size)
	if not minX then
		if o then hideObj(o) end
		return false
	end
	if not o then
		o = {} -- Register first so a partially created overlay can be cleaned up.
		objs[m] = o
		makeObj(o)
	end
	local col = e.enemy == nil and T.Sub or (e.enemy and ENEMY_HID or T.Good)
	if e.enemy and (not P.visColor or visCache[m]) then col = ENEMY_VIS end
	o.box.Position, o.box.Size = UDim2.fromOffset(minX, minY), UDim2.fromOffset(maxX - minX, maxY - minY)
	o.stroke.Color, o.box.Visible = col, P.box
	local cx = (minX + maxX) / 2
	local pl = e.player or modelPlayer(m)
	o.name.Position = UDim2.fromOffset(cx, math.max(14, minY - 1))
	local label = pl and pl.DisplayName
	if not label then
		if e.enemy == true then label = "Enemy"
		elseif e.enemy == false then label = "Teammate"
		else label = "Unknown player" end
	end
	o.name.Text, o.name.TextColor3, o.name.Visible = label, col, P.names
	o.dist.Position = UDim2.fromOffset(cx, math.min(c.ViewportSize.Y - 12, maxY + 1))
	o.dist.Text, o.dist.TextColor3, o.dist.Visible = math.floor(dist * 0.28) .. "m", col, P.dist
	o.hl.Adornee, o.hl.OutlineColor, o.hl.Enabled = m, col, P.chams
	o.line.Visible = P.tracers
	if P.tracers then
		local from, to = Vector2.new(c.ViewportSize.X / 2, c.ViewportSize.Y), Vector2.new(cx, maxY)
		local d = to - from
		o.line.Size = UDim2.fromOffset(d.Magnitude, 1)
		o.line.Position = UDim2.fromOffset((from.X + to.X) / 2, (from.Y + to.Y) / 2)
		o.line.Rotation, o.line.BackgroundColor3 = math.deg(math.atan2(d.Y, d.X)), col
	end
	return true
end
local function updateEsp()
	local c = cam()
	P.drawn, P.espErr, P.renderFailed = 0, nil, 0
	if not c then
		for _, o in pairs(objs) do hideObj(o) end
		return
	end
	local seen = {}
	if P.esp then
		for _, e in ipairs(collectEsp()) do
			seen[e.m] = true
			local ok, drawn = pcall(drawModel, c, e)
			if ok then
				if drawn then P.drawn = P.drawn + 1 end
			else
				P.renderFailed = P.renderFailed + 1
				P.espErr = tostring(drawn)
				-- A stale/partially replicated model must not stop every other row.
				local o = objs[e.m]
				if o then killObj(o) objs[e.m] = nil end
			end
		end
	end
	for m, o in pairs(objs) do
		if not seen[m] or not m.Parent then killObj(o) objs[m] = nil end
	end
end
bind(RunService.RenderStepped, function()
	local ok, err = pcall(updateEsp)
	if not ok then P.espErr = tostring(err) end
end)
